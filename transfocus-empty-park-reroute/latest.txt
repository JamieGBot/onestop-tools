 * ============================================================================
 * JAMIE — STANDALONE NIGHT SHIFT EMPTY PARK REROUTE V8 — FINAL I-MT-DEH
 * ============================================================================
 *
 * RUN THIS SNIPPET ON A TRANSFOCUS JOB PAGE.
 * THERE IS NO BUTTON — RUNNING THE SNIPPET PERFORMS THE ACTION IMMEDIATELY.
 *
 * BEFORE
 * ------
 *   I-MT-PU-DEH
 *   CLIENT  ->  EMPTY PARK
 *
 * AFTER
 * -----
 *   I-MT-PU
 *   CLIENT  ->  MEDLOG Rous Head Depot
 *
 *   I-MT-DEH
 *   MEDLOG Rous Head Depot  ->  SAME EMPTY PARK
 *
 * EXACT V5 SEQUENCE
 * -----------------
 * 1. Opens Job Legs.
 * 2. Finds and LOCKS the EXISTING I-MT-PU-DEH BEFORE adding a row.
 * 3. Reads its original CLIENT From + EMPTY PARK To.
 * 4. Clicks Transfocus + Add New Leg.
 * 5. Captures the newly-created default row.
 * 6. Moves that new row directly ABOVE the locked I-MT-PU-DEH.
 *    ALSO rewrites Transfocus data-leg-ordernum values so the order SAVES.
 * 7. New row Type = I-MT-PU.
 * 8. New row From = original CLIENT.
 * 9. New row To = MEDLOG Rous Head Depot.
 * 10. Copies Depot / Service Code / Leg Date from the dehire row where possible.
 * 11. EXISTING I-MT-PU-DEH Type = I-MT-DEH.
 * 12. EXISTING dehire From = MEDLOG Rous Head Depot.
 * 13. EXISTING dehire To remains the SAME EMPTY PARK.
 * 14. Finds the real Job Legs horizontal scrollbar and automatically scrolls
 *     until the MANIFEST column is clearly visible.
 *
 * AUTO-SCROLL TO MANIFEST
 * -----------------------
 * - Does NOT guess a fixed pixel distance anymore.
 * - Finds the actual "Manifest" column in the live Job Legs table.
 * - Finds the nearest real horizontal scroll owner for that table.
 * - Positions Manifest at roughly 58% across the visible Job Legs area.
 * - Works with the normal Transfocus table wrapper (.table-scrollable /
 *   .table-responsive) and DevExtreme if that layout is ever used.
 * - Re-applies after reroute/page handlers because scrollIntoView can move the
 *   horizontal position back.
 *
 * SAFETY
 * ------
 * - Existing I-MT-PU-DEH To / dehire destination is never changed.
 * - Does NOT click the final normal Job Save button.
 * - Stops rather than guessing if more than one I-MT-PU-DEH exists.
 * - If an I-MT-PU CLIENT -> ROUS leg already sits immediately above the
 *   dehire row, it removes the temporary new default row and does nothing else.
 * ============================================================================
 */

    const NAME =

    const VERSION =
        "8.0";

    const STATE_KEY =
        "__JAMIE_NIGHT_SHIFT_EMPTY_PARK_REROUTE_V8__";

    const ROUS_HEAD = {

        text:
            "MEDLOG Rous Head Depot"
    };

    /*
     * V7:
     * Where the centre of the Manifest column should sit inside the visible

     */
    const MANIFEST_VIEWPORT_RATIO =
        0.58;

    if (
        !String(location.hostname || "")
            .toLowerCase()
            .endsWith("tmspanel.com.au")
    ) {
        console.info(
            `[${NAME}] Not running — this is not Transfocus.`

    }

     */
    window[STATE_KEY] = {

            null,

            null

        new Promise(resolve =>

    function clean(value) {
        return String(
            value ?? ""
        )
            .replace(
                /[\u200B-\u200D\uFEFF]/g,
                ""
            )
            .replace(
                /\u00a0/g,
                " "
            )
            .replace(
                /\s+/g,
                " "
            )
            .trim();
    }

    function upper(value) {
        return clean(value)
            .toUpperCase();
    }

    function normalType(value) {
        return upper(value)
            .replace(
                /[–—]/g,
                "-"
            )
            .replace(
                /\s+/g,
                ""
            );
    }

    function uniqueElements(
        elements
    ) {
        return Array.from(
            new Set(
                elements.filter(
                    element =>
                        element instanceof
                        HTMLElement
                )
            )
        );
    }

    function jobLegRoots() {
        return [
            document.getElementById(
                "job-legs"
            ),

                "job-legs-panel"
            ),
            document.getElementById(
                "job-legs-grid"
            )

    }

    function findManifestColumn() {
        const roots =
            jobLegRoots();

        /*
         * Header first — this is the most stable target.
         */
        for (
            const root of
            roots
        ) {
            const headers =
                Array.from(
                    root.querySelectorAll(
                        "thead th, thead td"
                    )
                );

            const exact =
                headers.find(
                    cell =>
                        upper(
                            cell.textContent
                        ) ===
                        "MANIFEST"
                );

            if (exact) {
                return exact;
            }

            const partial =
                headers.find(
                    cell =>
                        upper(
                            cell.textContent
                        ).includes(
                            "MANIFEST"
                        )
                );

            if (partial) {
                return partial;
            }
        }

        /*
         * Fallback to a data-leg cell if Transfocus changes header markup.
         */
        for (
            const root of
            roots
        ) {
            const explicit =
                root.querySelector(
                    [
                        'td[data-leg="Manifest"]',
                        'th[data-leg="Manifest"]',
                        '[data-leg="Manifest"]',
                        '[data-id="Manifest"]'
                    ].join(",")
                );

            if (explicit) {
                return explicit;
            }
        }

        return null;
    }

    function computedHorizontalOverflow(
        element
    ) {
        try {
            const style =
                getComputedStyle(
                    element
                );

            return String(
                style.overflowX ||
                style.overflow ||
                ""
            ).toLowerCase();
        } catch (_) {
            return "";
        }
    }

    function isRealHorizontalScroller(
        element
    ) {
        if (
            !(
                element instanceof
                HTMLElement
            )
        ) {
            return false;
        }

        if (
            element.scrollWidth <=
            element.clientWidth + 8
        ) {
            return false;
        }

        const overflow =
            computedHorizontalOverflow(
                element
            );

        const classHint =
            /(?:table-scrollable|table-responsive|scrollable|scroll-body|datatables_scrollbody|dx-scrollable)/i.test(
                String(
                    element.className ||
                    ""
                )
            );

        return (
            /auto|scroll|overlay/i.test(
                overflow
            ) ||
            classHint
        );
    }

    function findManifestScrollOwner(
        manifestColumn
    ) {
        if (!manifestColumn) {
            return null;
        }

        /*
         * The real scrollbar must be an ancestor of the Manifest column.
         * Walk nearest -> furthest so we grab the table's own scrollbar rather
         * than the page/body.
         */
        let node =
            manifestColumn.parentElement;

        for (
            let depth = 0;
            node &&
            depth < 14;
            depth += 1
        ) {
            if (
                isRealHorizontalScroller(
                    node
                )
            ) {
                return node;
            }

            if (
                node.id ===
                "job-legs-panel"
            ) {
                break;
            }

            node =
                node.parentElement;
        }

        /*
         * Known Transfocus wrappers. This specifically covers the horizontal
         * scrollbar visible in the user's regular Job Legs table.
         */
        for (
            const root of
            jobLegRoots()
        ) {
            const known =
                Array.from(
                    root.querySelectorAll(
                        [
                            ".table-scrollable",
                            ".table-responsive",
                            ".dataTables_scrollBody",
                            ".dx-scrollable-container",
                            ".dx-datagrid-rowsview"
                        ].join(",")
                    )
                ).find(
                    element =>
                        element.contains(
                            manifestColumn
                        ) &&
                        element.scrollWidth >
                            element.clientWidth + 8
                );

            if (known) {
                return known;
            }
        }

        /*
         * Last fallback: scan ancestors by measurable horizontal overflow even
         * if their CSS overflow property is unusual.
         */
        node =
            manifestColumn.parentElement;

        for (
            let depth = 0;
            node &&
            depth < 14;
            depth += 1
        ) {
            if (
                node.scrollWidth >
                node.clientWidth + 8
            ) {
                return node;
            }

            node =
                node.parentElement;
        }

        return null;
    }

    function devExtremeScrollToManifest(
        manifestColumn
    ) {
        const grid =
            document.getElementById(
                "job-legs-grid"
            );

        if (
            !grid ||
            !window.jQuery ||
            typeof window.jQuery.fn
                ?.dxDataGrid !==
                "function"
        ) {
            return false;
        }

        try {
            const instance =
                window.jQuery(
                    grid
                ).dxDataGrid(
                    "instance"
                );

            const scrollable =
                instance
                    ?.getScrollable?.();

            if (!scrollable) {
                return false;
            }

            const container =
                grid.querySelector(
                    ".dx-datagrid-rowsview .dx-scrollable-container"
                ) ||
                grid.querySelector(
                    ".dx-datagrid-rowsview"
                ) ||
                grid;

            const columnRect =
                manifestColumn
                    .getBoundingClientRect();

            const viewportRect =
                container
                    .getBoundingClientRect();

            const currentLeft =
                Number(
                    scrollable
                        .scrollLeft?.() ||
                    0
                );

            const scrollWidth =
                Number(
                    scrollable
                        .scrollWidth?.() ||
                    0
                );

            const clientWidth =
                Number(
                    scrollable
                        .clientWidth?.() ||
                    viewportRect.width ||
                    0
                );

            const maxLeft =
                Math.max(
                    0,
                    scrollWidth -
                    clientWidth
                );

            const columnCentre =
                columnRect.left +
                columnRect.width / 2;

            const wantedScreenX =
                viewportRect.left +
                viewportRect.width *
                    MANIFEST_VIEWPORT_RATIO;

            const delta =
                columnCentre -
                wantedScreenX;

            const target =
                Math.max(
                    0,
                    Math.min(
                        maxLeft,
                        currentLeft +
                        delta
                    )
                );

            scrollable.scrollTo({
                left:
                    target,
                top:
                    Number(
                        scrollable
                            .scrollTop?.() ||
                        0
                    )
            });

            return true;
        } catch (_) {
            return false;
        }
    }

    function nativeScrollToManifest(
        manifestColumn,
        scroller
    ) {
        if (
            !manifestColumn ||
            !scroller
        ) {
            return false;
        }

        const columnRect =
            manifestColumn
                .getBoundingClientRect();

        const viewportRect =
            scroller
                .getBoundingClientRect();

        if (
            viewportRect.width <= 0
        ) {
            return false;
        }

        const currentLeft =
            Number(
                scroller.scrollLeft ||
                0
            );

        const maxLeft =
            Math.max(
                0,
                scroller.scrollWidth -
                scroller.clientWidth
            );

        const columnCentre =
            columnRect.left +
            columnRect.width / 2;

        const wantedScreenX =
            viewportRect.left +
            viewportRect.width *
                MANIFEST_VIEWPORT_RATIO;

        const delta =
            columnCentre -
            wantedScreenX;

        const target =
            Math.max(
                0,
                Math.min(
                    maxLeft,
                    currentLeft +
                    delta
                )
            );

        if (
            Math.abs(
                target -
                currentLeft
            ) < 2
        ) {
            return true;
        }

        try {
            scroller.scrollTo({
                left:
                    target,
                top:
                    scroller.scrollTop,
                behavior:
                    "auto"
            });
        } catch (_) {
            scroller.scrollLeft =
                target;
        }

        /*
         * A couple of Transfocus wrappers listen for the native scroll event
         * to synchronise headers/other UI.
         */
        try {
            scroller.dispatchEvent(
                new Event(
                    "scroll",
                    {
                        bubbles:
                            true
                    }
                )
            );
        } catch (_) {}

        return true;
    }

    function scrollJobLegsToManifest() {
        const manifestColumn =
            findManifestColumn();

        if (!manifestColumn) {
            return false;
        }

        /*
         * DevExtreme layout, if present.
         */
        if (
            devExtremeScrollToManifest(
                manifestColumn
            )
        ) {
            return true;
        }

        /*
         * Normal Transfocus Job Legs table — this is the layout shown in the
         * user's screenshot.
         */
        const scroller =
            findManifestScrollOwner(
                manifestColumn
            );

        if (!scroller) {
            return false;
        }

        return nativeScrollToManifest(
            manifestColumn,
            scroller
        );
    }

        /*
         * Keep the old function name because the reroute flow already calls it
         * in the correct places. V7's implementation now targets MANIFEST,
         * rather than a guessed fixed scrollLeft.

        [
            0,
            80,
            180,
            350,
            650,
            1100

            delay =>

                    scrollJobLegsToManifest,

                )
        );
    }

    async function ensureJobLegsOpen() {
        const link =
            document.getElementById(
                "job-legs-panel-link"
            );

        const panel =
            document.getElementById(
                "job-legs-panel"
            );

        if (!link || !panel) {
            return false;
        }

        const isOpen = () =>
            panel.classList.contains(
                "active"
            ) &&
            (
                panel.classList.contains(
                    "in"
                ) ||
                panel.offsetParent !== null
            );

        if (isOpen()) {
            scheduleJobLegsRightShift();
            return true;
        }

        const $ =
            window.jQuery;

        if (
            $ &&
            typeof $.fn?.tab ===
                "function"
        ) {
            $(link).tab("show");
        } else {
            link.click();
        }

        const started =
            performance.now();

        while (
            performance.now() -
                started <
            2200
        ) {
            if (isOpen()) {
                await sleep(80);
                scheduleJobLegsRightShift();
                return true;
            }

            await sleep(30);
        }

        return false;
    }

    function findBookingInitObjectText() {
        const script =
            Array.from(
                document.scripts
            ).find(element =>
                String(
                    element.textContent ||
                    ""
                ).includes(
                    "Booking.Init("
                )
            );

        if (!script) {
            return "";
        }

        const source =
            String(
                script.textContent ||
                ""
            );

        const marker =
            source.indexOf(
                "Booking.Init("
            );

        const start =
            source.indexOf(
                "{",
                marker
            );

        if (start < 0) {
            return "";
        }

        let depth = 0;
        let inString = false;
        let escaped = false;

        for (
            let index = start;
            index < source.length;
            index += 1
        ) {
            const character =
                source[index];

            if (inString) {
                if (escaped) {
                    escaped = false;
                    continue;
                }

                if (character === "\\") {
                    escaped = true;
                    continue;
                }

                if (character === '"') {
                    inString = false;
                }

                continue;
            }

            if (character === '"') {
                inString = true;
                continue;
            }

            if (character === "{") {
                depth += 1;
                continue;
            }

            if (character === "}") {
                depth -= 1;

                if (depth === 0) {
                    return source.slice(
                        start,
                        index + 1
                    );
                }
            }
        }

        return "";
    }

    function itemId(item) {
        return String(
            item?.id ??
            item?.ID ??
            item?.value ??
            item?.Value ??
            item?.Code ??
            ""
        );
    }

    function itemText(item) {
        return String(
            item?.text ??
            item?.Text ??
            item?.Name ??
            item?.Description ??
            item?.Code ??
            ""
        );
    }

    function findDropdownOption(
        listName,
        wantedText,
        exactOnly = false
    ) {
        const list =
            state.dropdowns?.[
                listName
            ];

        if (!Array.isArray(list)) {
            return null;
        }

        const wanted =
            upper(wantedText);

        const exact =
            list.find(item =>
                upper(
                    itemText(item)
                ) === wanted
            );

        if (exact) {
            return {
                id:
                    itemId(exact),
                text:
                    itemText(exact)
            };
        }

        if (exactOnly) {
            return null;
        }

        const partial =
            list.find(item =>
                upper(
                    itemText(item)
                ).includes(
                    wanted
                )
            );

        return partial
            ? {
                id:
                    itemId(partial),
                text:
                    itemText(partial)
            }
            : null;
    }

    function getRows() {
        const root =
            document.getElementById(
                "job-legs"
            );

        if (!root) {
            return [];
        }

        const rows =
            Array.from(
                root.querySelectorAll(
                    "tr.job-leg-row, tbody tr, tr"
                )
            );

        /*
         * A real Job Leg row has a JobLegTypeID control or a Type cell.
         * This avoids depending on one CSS class that Transfocus may drop
         * when it rebuilds editable rows.
         */
        return rows.filter(row =>
            !!(
                row.querySelector(
                    'input[data-id="JobLegTypeID"]'
                ) ||
                row.querySelector(
                    'td[data-leg="Type"]'
                )
            )
        );
    }

    function fieldInput(
        row,
        dataId
    ) {
        return row?.querySelector(
            `input[data-id="${dataId}"]`
        ) || null;
    }

    function cellFor(
        row,
        cellName,
        dataId = ""
    ) {
        return (
            row?.querySelector(
                `td[data-leg="${cellName}"]`
            ) ||
            fieldInput(
                row,
                dataId
            )?.closest("td") ||
            null
        );
    }

    function visibleFieldText(
        row,
        cellName,
        dataId
    ) {
        const input =
            fieldInput(
                row,
                dataId
            );

        const cell =
            cellFor(
                row,
                cellName,
                dataId
            );

        const candidates = [
            cell
                ?.querySelector(
                    ".select2-chosen"
                )
                ?.textContent,
            input
                ?.getAttribute(
                    "data-text"
                ),
            input
                ?.getAttribute(
                    "title"
                ),
            cell
                ?.querySelector(
                    "select option:checked"
                )
                ?.textContent,
            cell
                ?.innerText,
            cell
                ?.textContent
        ];

        for (const candidate of candidates) {
            const text =
                clean(candidate);

            if (
                text &&
                upper(text) !==
                    "SELECT A VALUE"
            ) {
                return text;
            }
        }

        return "";
    }

    function readField({
        row,
        cellName,
        dataId,
        listName = ""
    }) {
        const input =
            fieldInput(
                row,
                dataId
            );

        const text =
            visibleFieldText(
                row,
                cellName,
                dataId
            );

        let id =
            clean(
                input
                    ?.getAttribute(
                        "data-value"
                    ) ||
                input?.value ||
                ""
            );

        let resolvedText =
            text;

        if (
            listName &&
            resolvedText
        ) {
            const option =
                findDropdownOption(
                    listName,
                    resolvedText,
                    true
                );

            if (option) {
                id =
                    id ||
                    option.id;

                resolvedText =
                    option.text;
            }
        }

        if (!id && !resolvedText) {
            return null;
        }

        return {
            id,
            text:
                resolvedText
        };
    }

    function rowTypeCandidates(row) {
        const input =
            fieldInput(
                row,
                "JobLegTypeID"
            );

        const cell =
            cellFor(
                row,
                "Type",
                "JobLegTypeID"
            );

        const candidates = [
            cell
                ?.querySelector(
                    ".select2-chosen"
                )
                ?.textContent,
            input
                ?.getAttribute(
                    "data-text"
                ),
            input
                ?.getAttribute(
                    "title"
                ),
            cell
                ?.querySelector(
                    "select option:checked"
                )
                ?.textContent,
            input?.value,
            cell?.innerText,
            cell?.textContent,
            row?.innerText,
            row?.textContent
        ]
            .map(clean)
            .filter(Boolean);

        return Array.from(
            new Set(candidates)
        );
    }

    function rowTypeText(row) {
        const candidates =
            rowTypeCandidates(row);

        const exactLooking =
            candidates.find(value =>
                /I\s*[-–—]?\s*MT\s*[-–—]?\s*PU/i.test(
                    value
                )
            );

        return exactLooking ||
            candidates[0] ||
            "";
    }

    function isDehireType(text) {
        const type =
            normalType(text);

        return (
            type ===
                "I-MT-PU-DEH" ||
            type.includes(
                "I-MT-PU-DEH"
            ) ||
            type.endsWith(
                "MT-PU-DEH"
            )
        );
    }

    function isDehireRow(row) {
        return rowTypeCandidates(row)
            .some(isDehireType);
    }

    function dehireRows(
        excludeRow = null
    ) {
        return getRows()
            .filter(row =>
                row !== excludeRow &&
                isDehireRow(row)
            );
    }

    function dehireFingerprint(row) {
        const from =
            readField({
                row,
                cellName: "From",
                dataId: "SenderID",
                listName: "Locations"
            });

        const to =
            readField({
                row,
                cellName: "To",
                dataId: "RecepientID",
                listName: "Locations"
            });

        const jobLeg =
            clean(
                row.querySelector(
                    'td[data-leg="Job Leg Number"], td[data-leg="JobLegNumber"]'
                )?.textContent ||
                row.querySelector(
                    'input[data-id="JobLegNumber"]'
                )?.value ||
                ""
            );

        return {
            from,
            to,
            jobLeg
        };
    }

    function refindLockedDehire(
        lockedRow,
        fingerprint,
        excludeRow = null
    ) {
        if (
            lockedRow?.isConnected &&
            lockedRow !== excludeRow &&
            isDehireRow(lockedRow)
        ) {
            return lockedRow;
        }

        const rows =
            dehireRows(
                excludeRow
            );

        if (rows.length === 1) {
            return rows[0];
        }

        if (!rows.length) {
            return null;
        }

        const exact =
            rows.filter(row => {
                const candidate =
                    dehireFingerprint(row);

                const sameFrom =
                    fingerprint?.from &&
                    candidate.from &&
                    valuesEqual(
                        candidate.from,
                        fingerprint.from
                    );

                const sameTo =
                    fingerprint?.to &&
                    candidate.to &&
                    valuesEqual(
                        candidate.to,
                        fingerprint.to
                    );

                const sameJobLeg =
                    clean(
                        fingerprint?.jobLeg
                    ) &&
                    clean(
                        candidate.jobLeg
                    ) ===
                    clean(
                        fingerprint.jobLeg
                    );

                return (
                    (sameFrom && sameTo) ||
                    (sameJobLeg && sameTo)
                );
            });

        return exact.length === 1
            ? exact[0]
            : null;
    }

    async function waitForNewRow(
        previousRows,
        timeout = 4500
    ) {
        const started =
            performance.now();

        while (
            performance.now() -
                started <
            timeout
        ) {
            const row =
                getRows().find(item =>
                    !previousRows.has(item)
                );

            if (row) {
                await sleep(100);
                return row;
            }

            await sleep(25);
        }

        return null;
    }

    function resolveExistingField({
        row,
        cellName,
        dataId,
        listName
    }) {
        const current =
            readField({
                row,
                cellName,
                dataId,
                listName
            });

        if (!current?.text) {
            return null;
        }

        if (current.id) {
            return current;
        }

        return (
            findDropdownOption(
                listName,
                current.text,
                true
            ) ||
            findDropdownOption(
                listName,
                current.text,
                false
            )
        );
    }

    function removeOldSelect2Container(
        input
    ) {
        const cell =
            input.closest("td");

        if (!cell) {
            return;
        }

        Array.from(
            cell.children
        ).forEach(child => {
            if (
                child !== input &&
                child.classList
                    ?.contains(
                        "select2-container"
                    )
            ) {
                child.remove();
            }
        });
    }

    function listForField(
        listName,
        selected
    ) {
        if (
            listName ===
            "Locations"
        ) {
            return [
                selected
            ];
        }

        const original =
            state.dropdowns?.[
                listName
            ];

        if (
            Array.isArray(original) &&
            original.length
        ) {
            return original.map(item => ({
                id:
                    itemId(item),
                text:
                    itemText(item)
            }));
        }

        return [
            selected
        ];
    }

    function initialiseSelect2(
        input,
        list
    ) {
        const $ =
            window.jQuery;

        if (
            !$ ||
            typeof $.fn?.select2 !==
                "function"
        ) {
            throw new Error(
                "Transfocus Select2 is unavailable."
            );
        }

        const $input =
            $(input);

        if (
            input.classList.contains(
                "select2-offscreen"
            )
        ) {
            try {
                $input.select2(
                    "destroy"
                );
            } catch (_) {}
        }

        removeOldSelect2Container(
            input
        );

        input.classList.remove(
            "lazy-table-select2",
            "lazy-jobleg-select2"
        );

        $input.select2({
            data:
                list,
            width:
                "resolve",
            allowClear:
                false,
            placeholder:
                "Select a value"
        });

        return input;
    }

    function setSelectField({
        row,
        dataId,
        listName,
        option,
        triggerChange = true
    }) {
        const $ =
            window.jQuery;

        if (!$) {
            throw new Error(
                "Transfocus jQuery is unavailable."
            );
        }

        if (
            !clean(option?.id) ||
            !clean(option?.text)
        ) {
            throw new Error(
                `${dataId} option could not be resolved.`
            );
        }

        let input =
            fieldInput(
                row,
                dataId
            );

        if (!input) {
            throw new Error(
                `${dataId} field was not found.`
            );
        }

        if (
            !input.classList.contains(
                "select2-offscreen"
            ) ||
            !input
                .closest("td")
                ?.querySelector(
                    ".select2-container"
                )
        ) {
            input =
                initialiseSelect2(
                    input,
                    listForField(
                        listName,
                        option
                    )
                );
        }

        const selected = {
            id:
                String(option.id),
            text:
                String(option.text)
        };

        const $input =
            $(input);

        try {
            $input.select2(
                "data",
                selected
            );
        } catch (_) {
            input =
                initialiseSelect2(
                    input,
                    listForField(
                        listName,
                        option
                    )
                );

            $(input).select2(
                "data",
                selected
            );
        }

        const finalInput =
            fieldInput(
                row,
                dataId
            ) || input;

        const $final =
            $(finalInput);

        $final.val(
            selected.id
        );

        finalInput.value =
            selected.id;

        finalInput.setAttribute(
            "value",
            selected.id
        );

        finalInput.setAttribute(
            "data-value",
            selected.id
        );

        finalInput.setAttribute(
            "data-text",
            selected.text
        );

        $final.attr(
            "data-value",
            selected.id
        );

        $final.attr(
            "data-text",
            selected.text
        );

        const chosen =
            finalInput
                .closest("td")
                ?.querySelector(
                    ".select2-chosen"
                );

        if (chosen) {
            chosen.textContent =
                selected.text;
        }

        if (triggerChange) {
            $final.trigger(
                "change"
            );
        }
    }

    function valuesEqual(
        current,
        wanted
    ) {
        if (!current || !wanted) {
            return false;
        }

        if (
            clean(current.id) &&
            clean(wanted.id) &&
            clean(current.id) ===
                clean(wanted.id)
        ) {
            return true;
        }

        return (
            upper(current.text) ===
            upper(wanted.text)
        );
    }

        const cell =
            orderCell(row);

        const value =

    function setOrderLikeInput(
        input,
        order
    ) {
        const signature =
            upper(
                [
                    input.getAttribute(
                        "data-id"
                    ),
                    input.getAttribute(
                        "data-field"
                    ),
                    input.getAttribute(
                        "name"
                    ),
                    input.id
                ].join(" ")
            );

        /*
         * Only touch fields that are clearly an ORDER / SEQUENCE field.
         * Never touch Job Leg Number or IDs.
         */
        if (
            !/(SORT\s*ORDER|DISPLAY\s*ORDER|JOB\s*LEG\s*ORDER|LEG\s*ORDER|SEQUENCE|POSITION)/i.test(
                signature
            ) ||
            /NUMBER|ID$/i.test(
                signature
            )
        ) {
            return false;
        }

        input.value =
            String(order);

        input.setAttribute(
            "value",
            String(order)
        );

        try {
            window.jQuery?.(input)
                ?.val?.(
                    String(order)
                );
        } catch (_) {}

        try {
            input.dispatchEvent(
                new Event(
                    "change",
                    {
                        bubbles: true
                    }
                )
            );
        } catch (_) {}

        return true;
    }

    function writeLegOrder(
        row,
        order
    ) {
        const cell =
            orderCell(row);

        if (cell) {
            cell.setAttribute(
                "data-leg-ordernum",
                String(order)
            );

            /*
             * jQuery caches data-* values. Keep that cache in sync too.
             */
            try {
                window.jQuery?.(cell)
                    ?.data?.(
                        "leg-ordernum",
                        order
                    );
            } catch (_) {}

            /*
             * The # column is normally plain text. If it has no child
             * controls, keep the visible number aligned with the saved order.
             */
            if (
                cell.children.length === 0
            ) {
                cell.textContent =
                    String(order);
            }
        }

        if (
            row.hasAttribute(
                "data-leg-ordernum"
            )
        ) {
            row.setAttribute(
                "data-leg-ordernum",
                String(order)
            );
        }

        row.querySelectorAll(
            "input"
        ).forEach(input => {
            setOrderLikeInput(
                input,
                order
            );
        });
    }

    function syncLegOrderMetadata(
        parent
    ) {
        const rows =
            getRows().filter(row =>
                row.parentElement ===
                    parent
            );

        rows.forEach(
            (row, index) => {
                writeLegOrder(
                    row,
                    index + 1
                );
            }
        );

        return rows;
    }

    function findSortableOwner(
        parent
    ) {
        const $ =
            window.jQuery;

        if (!$) {
            return null;
        }

        let node =
            parent;

        for (
            let depth = 0;
            node &&
            depth < 6;
            depth += 1
        ) {
            const $node =
                $(node);

            let instance =
                null;

            try {
                instance =
                    $node.sortable?.(
                        "instance"
                    ) ||
                    $node.data(
                        "ui-sortable"
                    ) ||
                    $node.data(
                        "sortable"
                    ) ||
                    null;
            } catch (_) {
                try {
                    instance =
                        $node.data(
                            "ui-sortable"
                        ) ||
                        $node.data(
                            "sortable"
                        ) ||
                        null;
                } catch (_) {}
            }

            if (
                instance ||
                node.classList
                    ?.contains(
                        "ui-sortable"
                    )
            ) {
                return {
                    node,
                    $node,
                    instance
                };
            }

            node =
                node.parentElement;
        }

        return null;
    }

    function notifyRealSortable(
        row,
        parent
    ) {
        const $ =
            window.jQuery;

        if (!$) {
            return false;
        }

        const owner =
            findSortableOwner(
                parent
            );

        const $row =
            $(row);

        const ui = {
            item: $row,
            helper: $row
        };

        if (owner) {
            try {
                owner.$node.sortable(
                    "refreshPositions"
                );
            } catch (_) {}

            try {
                owner.$node.sortable(
                    "refresh"
                );
            } catch (_) {}

            /*
             * This is the important V5 part: call jQuery UI Sortable's real
             * update hook, rather than merely firing a similarly-named DOM
             * event. Transfocus can then persist whatever order state its
             * own sortable callback maintains.
             */
            try {
                if (
                    typeof owner.instance
                        ?._trigger ===
                        "function"
                ) {
                    owner.instance._trigger(
                        "update",
                        $.Event(
                            "sortupdate"
                        ),
                        ui
                    );

                    owner.instance._trigger(
                        "stop",
                        $.Event(
                            "sortstop"
                        ),
                        ui
                    );

                    owner.$node.trigger(
                        "change"
                    );

                    return true;
                }
            } catch (_) {}

            try {
                const update =
                    owner.$node.sortable(
                        "option",
                        "update"
                    );

                if (
                    typeof update ===
                        "function"
                ) {
                    update.call(
                        owner.node,
                        $.Event(
                            "sortupdate"
                        ),
                        ui
                    );

                    owner.$node.trigger(
                        "change"
                    );

                    return true;
                }
            } catch (_) {}

            try {
                owner.$node.trigger(
                    "sortupdate",
                    [ui]
                );

                owner.$node.trigger(
                    "sortstop",
                    [ui]
                );

                owner.$node.trigger(
                    "change"
                );

                return true;
            } catch (_) {}
        }

        /*
         * Fallback: notify the actual Job Legs container and row parent.
         */
        for (
            const node of
            [
                parent,
                document.getElementById(
                    "job-legs"
                )
            ]
        ) {
            if (!node) continue;

            try {
                $(node).trigger(
                    "sortupdate",
                    [ui]
                );

                $(node).trigger(
                    "sortstop",
                    [ui]
                );

                $(node).trigger(
                    "change"
                );
            } catch (_) {}
        }

        return false;
    }

    async function moveImmediatelyAbove(
        row,
        targetRow
    ) {
        const parent =
            targetRow.parentElement;

        if (!parent) {
            throw new Error(
                "Could not access the Job Leg row container."
            );
        }

        /*
         * DOM order first.
         */
        parent.insertBefore(
            row,
            targetRow
        );

        /*
         * Then rewrite Transfocus's REAL leg-order metadata.
         */
        syncLegOrderMetadata(
            parent
        );

        notifyRealSortable(
            row,
            parent
        );

        /*
         * Give page handlers a moment, then force metadata back into line in
         * case a handler reused an old data-leg-ordernum value.
         */
        await sleep(120);

        if (
            row.isConnected &&
            targetRow.isConnected &&
            targetRow.parentElement ===
                parent
        ) {
            if (
                row.nextElementSibling !==
                    targetRow
            ) {
                parent.insertBefore(
                    row,
                    targetRow
                );
            }

            syncLegOrderMetadata(
                parent
            );
        }
    }

    function isAlreadyRouted(
        dehireRow
    ) {
        const dehireFrom =
            rowField(
                dehireRow,
                "From",
                "SenderID",
                "Locations"
            );

        const previous =
            dehireRow
                .previousElementSibling;

        if (!previous) {
            return false;
        }

        const previousType =
            normalType(
                rowTypeText(
                    previous
                )
            );

        const previousFrom =
            rowField(
                previous,
                "From",
                "SenderID",
                "Locations"
            );

        const previousTo =
            rowField(
                previous,
                "To",
                "RecepientID",
                "Locations"
            );

        /*
         * V5 intended final state:
         *
         *   I-MT-PU   CLIENT -> ROUS HEAD
         *   I-MT-DEH  ROUS HEAD -> original dehire park
         *
         * Therefore the duplicate check is based on BOTH rows joining at
         * Rous Head. The pickup's From is intentionally NOT the same as the
         * dehire's From after routing.
         */
        return (
            previousType ===
                "I-MT-PU" &&
            valuesEqual(
                previousTo,
                ROUS_HEAD
            ) &&
            valuesEqual(
                dehireFrom,
                ROUS_HEAD
            )
        );
    }

    async function run() {
        let newRow =
            null;

        let dehireRow =
            null;

        try {
            const opened =
                await ensureJobLegsOpen();

            if (!opened) {
                throw new Error(
                    "Could not open Job Legs."
                );
            }

            scheduleJobLegsRightShift();

            state.dropdowns =
                readDropdownData();

            /*
             * V4 KEY FIX:
             * Lock the EXISTING dehire BEFORE Add New Leg.
             */
            const dehiresBefore =
                dehireRows();

            console.info(
                `[${NAME}] Existing Job Leg types:`,
                getRows().map(row =>
                    rowTypeCandidates(row)
                )
            );

            if (!dehiresBefore.length) {
                throw new Error(
                    "No existing I-MT-PU-DEH leg was found BEFORE adding the new leg."
                );
            }

            if (
                dehiresBefore.length > 1
            ) {
                throw new Error(
                    `Found ${dehiresBefore.length} I-MT-PU-DEH legs. Stopped so the wrong one is not changed.`
                );
            }

            dehireRow =
                dehiresBefore[0];

            state.dehireRow =
                dehireRow;

            const fingerprint =
                dehireFingerprint(
                    dehireRow
                );

            const originalClient =
                rowField(
                    dehireRow,
                    "From",
                    "SenderID",
                    "Locations"
                );

            const originalDestination =
                rowField(
                    dehireRow,
                    "To",
                    "RecepientID",
                    "Locations"
                );

            const depot =
                rowField(
                    dehireRow,
                    "Depot",
                    "DepotID",
                    "DepotCodes"
                );

            const service =
                rowField(
                    dehireRow,
                    "Service Code",
                    "ServiceCodeID",
                    "ServiceCodes"
                );

            const legDate =
                readLegDate(
                    dehireRow
                );

            if (
                !originalClient?.id ||
                !originalClient?.text
            ) {
                throw new Error(
                    `Could not reliably read the CLIENT From value from I-MT-PU-DEH. Visible value: ${originalClient?.text || "blank"}`
                );
            }

            if (
                !originalDestination?.id ||
                !originalDestination?.text
            ) {
                throw new Error(
                    `Could not reliably read the dehire destination To value from I-MT-PU-DEH. Visible value: ${originalDestination?.text || "blank"}`
                );
            }

            /*
             * Already correct?
             * Existing dehire starts at Rous Head AND the row directly above
             * is I-MT-PU ending at Rous Head.
             */
            if (
                isAlreadyRouted(
                    dehireRow
                )
            ) {
                console.info(
                    `[${NAME}] Already routed via Rous Head. Nothing changed.`
                );

                dehireRow.scrollIntoView({
                    behavior: "smooth",
                    block: "center",
                    inline: "nearest"
                });

                scheduleJobLegsRightShift();

                return;
            }

            const addButton =
                document.getElementById(
                    "btn-leg-add"
                );

            if (!addButton) {
                throw new Error(
                    "+ Add New Leg button was not found."
                );
            }

            const before =
                new Set(
                    getRows()
                );

            console.info(
                `[${NAME}] Locked I-MT-PU-DEH: ${originalClient.text} -> ${originalDestination.text}`
            );

            console.info(
                `[${NAME}] Clicking + Add New Leg…`
            );

            addButton.click();

            newRow =
                await waitForNewRow(
                    before
                );

            if (!newRow) {
                throw new Error(
                    "Transfocus did not create the new default Job Leg row."
                );
            }

            state.newRow =
                newRow;

            /*
             * If Add New Leg rebuilt old rows, relocate the dehire using the
             * values we locked before the click.
             */
            dehireRow =
                refindLockedDehire(
                    dehireRow,
                    fingerprint,
                    newRow
                );

            if (!dehireRow) {
                throw new Error(
                    "The I-MT-PU-DEH was locked before Add New Leg, but Transfocus rebuilt the table and V4 could not uniquely relocate that same dehire row."
                );
            }

            state.dehireRow =
                dehireRow;

            const pickupType =
                findDropdownOption(
                    "JobLegTypes",
                    "I-MT-PU",
                    true
                ) ||
                findDropdownOption(
                    "JobLegTypes",
                    "I-MT-PU"
                );

            if (!pickupType?.id) {
                throw new Error(
                    "I-MT-PU could not be found in the Transfocus Job Leg Type dropdown."
                );
            }

            const finalDehireType =
                findDropdownOption(
                    "JobLegTypes",
                    "I-MT-DEH",
                    true
                ) ||
                findDropdownOption(
                    "JobLegTypes",
                    "I-MT-DEH"
                );

            if (!finalDehireType?.id) {
                throw new Error(
                    "I-MT-DEH could not be found in the Transfocus Job Leg Type dropdown."
                );
            }

            const verified =
                findDropdownOption(
                    "JobLegStatuses",
                    "Verified",
                    true
                ) ||
                findDropdownOption(
                    "JobLegStatuses",
                    "Verified"
                );

            /*
             * Put new pickup directly above the existing dehire first.
             */
            await moveImmediatelyAbove(
                newRow,
                dehireRow
            );

            await sleep(80);

            /*
             * NEW LEG:
             * I-MT-PU
             * CLIENT -> MEDLOG Rous Head Depot
             */
            setSelectField({
                row: newRow,
                dataId: "JobLegTypeID",
                listName: "JobLegTypes",
                option: pickupType,
                triggerChange: true
            });

            await sleep(70);

            if (depot?.id) {
                setSelectField({
                    row: newRow,
                    dataId: "DepotID",
                    listName: "DepotCodes",
                    option: depot,
                    triggerChange: true
                });
            }

            setSelectField({
                row: newRow,
                dataId: "SenderID",
                listName: "Locations",
                option: originalClient,
                triggerChange: true
            });

            setSelectField({
                row: newRow,
                dataId: "RecepientID",
                listName: "Locations",
                option: ROUS_HEAD,
                triggerChange: true
            });

            if (service?.id) {
                setSelectField({
                    row: newRow,
                    dataId: "ServiceCodeID",
                    listName: "ServiceCodes",
                    option: service,
                    triggerChange: true
                });
            }

            if (verified?.id) {
                setSelectField({
                    row: newRow,
                    dataId: "JobLegStatusID",
                    listName: "JobLegStatuses",
                    option: verified,
                    triggerChange: true
                });
            }

            setLegDate(
                newRow,
                legDate
            );

            setBookedContainerStatus(
                newRow
            );

            /*
             * EXISTING DEHIRE:
             * I-MT-DEH
             * ROUS HEAD -> SAME EMPTY PARK
             */
            setSelectField({
                row: dehireRow,
                dataId: "JobLegTypeID",
                listName: "JobLegTypes",
                option: finalDehireType,
                triggerChange: true
            });

            await sleep(70);

            setSelectField({
                row: dehireRow,
                dataId: "SenderID",
                listName: "Locations",
                option: ROUS_HEAD,
                triggerChange: true
            });

            setSelectField({
                row: dehireRow,
                dataId: "RecepientID",
                listName: "Locations",
                option: originalDestination,
                triggerChange: true
            });

            if (verified?.id) {
                setSelectField({
                    row: dehireRow,
                    dataId: "JobLegStatusID",
                    listName: "JobLegStatuses",
                    option: verified,
                    triggerChange: true
                });
            }

            setBookedContainerStatus(
                dehireRow
            );

            /*
             * Final order repair after field change handlers have finished.
             */
            await sleep(180);

            if (
                newRow.isConnected &&
                dehireRow.isConnected &&
                newRow.parentElement ===
                    dehireRow.parentElement
            ) {
                const parent =
                    dehireRow.parentElement;

                if (
                    newRow.nextElementSibling !==
                        dehireRow
                ) {
                    parent.insertBefore(
                        newRow,
                        dehireRow
                    );
                }

                syncLegOrderMetadata(
                    parent
                );

                notifyRealSortable(
                    newRow,
                    parent
                );
            }

            highlightRows(
                newRow,
                dehireRow
            );

            newRow.scrollIntoView({
                behavior: "smooth",
                block: "center",
                inline: "nearest"
            });

            scheduleJobLegsRightShift();

            console.info(
                `[${NAME}] Complete. Review Job Legs, then use the normal Job Save button when ready.`
            );
        } catch (error) {
            console.error(
                `[${NAME}]`,
                error
            );

            /*
             * If we created a temporary default row and failed before a safe
             * reroute was completed, remove it rather than leaving junk behind.
             */
            if (
                newRow?.isConnected &&
                (
                    !dehireRow ||
                    !isAlreadyRouted(
                        dehireRow
                    )
                )
            ) {
                try {
                    const type =
                        normalType(
                            rowTypeText(
                                newRow
                            )
                        );

                    /*
                     * Only auto-remove if it still appears to be the temporary
                     * default / not the final pickup row.
                     */
                    if (
                        type !==
                        "I-MT-PU"
                    ) {
                        removeTemporaryRow(
                            newRow
                        );
                    }
                } catch (_) {}
            }

            alert(
                `Empty Park Reroute failed:\n${error?.message || error}`
            );
        } finally {
            state.busy =
                false;

            scheduleJobLegsRightShift();
        }
    }

    run();
})();
