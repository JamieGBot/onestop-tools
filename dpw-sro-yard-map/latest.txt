    function clean(
        value
    ) {

    }

    function addManagedEvent(
        target,
        type,
        handler,
        options
    ) {
        if (
            !target ||
            typeof target.addEventListener !==
                "function"
        ) {
            return handler;
        }

        target.addEventListener(
            type,
            handler,
            options
        );

        registerUnbinder(
            () => {
                try {
                    target.removeEventListener(
                        type,
                        handler,
                        options
                    );
                } catch (_) {}
            }
        );

        return handler;
    }

    function clamp(
        value,
        min,
        max
    ) {
        const n =
            Number(value);

        if (!Number.isFinite(n)) {
            return min;
        }

        return Math.min(
            max,
            Math.max(
                min,
                n
            )
        );
    }

    function getSroTable() {
        const holder =
            getHolder();

        if (!holder) {
            return null;
        }

        return Array.from(
            holder.querySelectorAll(
                "table"

        ).find(
            table =>

                table.querySelector(

                )
        ) || null;
    }

    function parseLocation(
        rawValue
    ) {
        const raw =
            clean(
                rawValue
            );

        if (!raw) {
            return null;
        }

        /*
         * Everything before the final 3 number groups is the DPW yard block.
         *
         * FRT-32-41-04-2
         * ^^^^^^  ^^ ^^ ^
         * block   X  Y  tier
         */
        const match =
            raw.match(
                /^(.*)-(\d+)-(\d+)-(\d+)$/
            );

        if (!match) {
            return null;
        }

        const block =
            clean(
                match[1]
            );

        const x =
            clean(
                match[2]
            );

        const y =
            clean(
                match[3]
            );

        const tier =
            Number(
                match[4]
            );

        if (
            !block ||
            !x ||
            !y ||
            !Number.isInteger(
                tier
            ) ||
            tier < 1
        ) {
            return null;
        }

        return {
            raw,
            block,
            x,
            y,
            tier,

            /*
             * Human-facing stack label.
             * Don't show "Column / Row" — just show the stack location.
             */
            stackLabel:
                `${x}-${y}`,

            stackKey:
                `${block}|${x}|${y}`
        };
    }

    function headerIndexes(
        table
    ) {
        const map =
            {};

        Array.from(
            table.querySelectorAll(
                "thead th"
            )
        ).forEach(
            (
                header,
                index
            ) => {
                const code =
                    clean(
                        header.getAttribute(
                            "data-code"
                        )
                    );

                if (code) {
                    map[
                        code
                    ] =
                        index;
                }
            }
        );

        return map;
    }

    function readCell(
        cells,
        indexes,
        code
    ) {
        const index =
            indexes[
                code
            ];

        if (
            !Number.isInteger(
                index
            ) ||
            !cells[
                index
            ]
        ) {
            return "";
        }

        return clean(
            cells[
                index
            ].textContent
        );
    }

    function readRecords() {
        const table =
            getSroTable();

        if (!table) {
            return [];
        }

        const indexes =
            headerIndexes(
                table
            );

        const tbody =
            table.tBodies?.[
                0
            ];

        if (!tbody) {
            return [];
        }

        return Array.from(
            tbody.rows
        )
            .map(
                rowElement => {
                    const cells =
                        Array.from(
                            rowElement.cells
                        );

                    const container =
                        readCell(
                            cells,
                            indexes,
                            "ContainerNumber"
                        );

                    if (!container) {
                        return null;
                    }

                    const yardLocation =
                        readCell(
                            cells,
                            indexes,
                            "YardLocation"
                        );

                    const checkbox =
                        rowElement.querySelector(
                            'input[name="srocontainersChk"]'
                        );

                    /*
                     * V33: DPW removes the Select checkbox once the container
                     * is no longer available on the live SRO (for example,
                     * already collected). Those historical rows remain visible
                     * in the table, but must NOT remain on the yard map.
                     */
                    if (!checkbox) {
                        return null;
                    }

                    return {
                        rowElement,
                        container,
                        priority:
                            readCell(
                                cells,
                                indexes,
                                "Priority"
                            ),
                        iso:
                            readCell(
                                cells,
                                indexes,
                                "IsoCode"
                            ),
                        commodity:
                            readCell(
                                cells,
                                indexes,
                                "Commodity"
                            ),
                        estDischarge:
                            readCell(
                                cells,
                                indexes,
                                "EstDischarge"
                            ),
                        yardLocation,
                        vessel:
                            readCell(
                                cells,
                                indexes,
                                "VesselName"
                            ),
                        voyage:
                            readCell(
                                cells,
                                indexes,
                                "Voyage"
                            ),
                        stops:
                            readCell(
                                cells,
                                indexes,
                                "Stops"
                            ),
                        status:
                            readCell(
                                cells,
                                indexes,
                                "Status"
                            ),
                        selectable:
                            true,
                        location:
                            parseLocation(
                                yardLocation
                            ),
                        above:
                            [],
                        accessState:
                            "unknown"
                    };
                }
            )
            .filter(Boolean);
    }

    function okayStatus(
        record
    ) {
        const status =
            upper(
                record.status
            );

        return (
            !status ||
            status === "OK"
        );
    }

    function analyse(
        records
    ) {
        const stacks =
            new Map();

        for (
            const record of
            records
        ) {
            if (
                !record.location
            ) {
                record.accessState =
                    upper(
                        record.stops
                    ).includes(
                        "HELD"
                    )
                        ? "held"
                        : "unlocated";

                continue;
            }

            const key =
                record.location
                    .stackKey;

            if (
                !stacks.has(
                    key
                )
            ) {
                stacks.set(
                    key,
                    []
                );
            }

            stacks
                .get(
                    key
                )
                .push(
                    record
                );
        }

        for (
            const stackRecords of
            stacks.values()
        ) {
            stackRecords.sort(
                (
                    a,
                    b
                ) =>
                    b.location
                        .tier -
                    a.location
                        .tier
            );

            for (
                const record of
                stackRecords
            ) {
                record.above =
                    stackRecords.filter(
                        other =>
                            other.location
                                .tier >
                            record.location
                                .tier
                    );

                if (
                    upper(
                        record.stops
                    ).includes(
                        "HELD"
                    )
                ) {
                    record.accessState =
                        "held";

                    continue;
                }

                if (
                    !okayStatus(
                        record
                    )
                ) {
                    record.accessState =
                        "issue";

                    continue;
                }

                if (
                    !record.selectable
                ) {
                    record.accessState =
                        "unavailable";

                    continue;
                }

                if (
                    record.above.length >
                    0
                ) {
                    record.accessState =
                        "blocked";
                } else if (
                    record.location
                        .tier ===
                    DPW_MAX_TIER
                ) {
                    /*
                     * Max 4 high means a Tier 4 container physically cannot
                     * have another container above it.
                     */
                    record.accessState =
                        "clear-above";
                } else {
                    /*
                     * No SRO container is visible above it, but another
                     * company's / another SRO's container may still be there.
                     */
                    record.accessState =
                        "top-of-sro";
                }
            }
        }

        return {
            records,
            stacks
        };
    }

    function majorityVessel(
        records
    ) {
        const counts =
            new Map();

        for (
            const record of
            records
        ) {
            const vessel =
                clean(
                    record.vessel
                );

            if (!vessel) {
                continue;
            }

            const voyage =
                clean(
                    record.voyage
                );

            const key =
                `${vessel}|${voyage}`;

            const current =
                counts.get(
                    key
                ) || {
                    vessel,
                    voyage,
                    count:
                        0
                };

            current.count +=
                1;

            counts.set(
                key,
                current
            );
        }

        return Array.from(
            counts.values()
        )
            .sort(
                (
                    a,
                    b
                ) =>
                    b.count -
                    a.count
            )[
                0
            ] || null;
    }

    function groupedData(
        records
    ) {
        const blocks =
            new Map();

        for (
            const record of
            records
        ) {
            if (
                !record.location
            ) {
                continue;
            }

            const block =
                record.location
                    .block;

            const key =
                record.location
                    .stackKey;

            if (
                !blocks.has(
                    block
                )
            ) {
                blocks.set(
                    block,
                    new Map()
                );
            }

            const stacks =
                blocks.get(
                    block
                );

            if (
                !stacks.has(
                    key
                )
            ) {
                stacks.set(
                    key,
                    {
                        key,
                        block,
                        x:
                            record.location
                                .x,
                        y:
                            record.location
                                .y,
                        label:
                            record.location
                                .stackLabel,
                        records:
                            []
                    }
                );
            }

            stacks
                .get(
                    key
                )
                .records
                .push(
                    record
                );
        }

        return blocks;
    }

    function stackTop(
        stack
    ) {
        return stack.records
            .slice()
            .sort(
                (
                    a,
                    b
                ) =>
                    b.location
                        .tier -
                    a.location
                        .tier
            )[
                0
            ] || null;
    }

    function stackState(
        stack
    ) {
        const top =
            stackTop(
                stack
            );

        if (!top) {
            return "empty";
        }

        return top.accessState;
    }

    function countState(
        records,
        stateName
    ) {
        return records.filter(
            record =>
                record.accessState ===

        ).length;
    }

    function blockStats(
        stacks
    ) {
        const all =
            Array.from(
                stacks.values()
            );

        const records =
            all.flatMap(
                stack =>
                    stack.records
            );

        return {
            stacks:
                all.length,
            containers:
                records.length,
            clearAbove:
                all.filter(
                    stack =>
                        stackState(
                            stack
                        ) ===
                        "clear-above"
                ).length,
            held:
                all.filter(
                    stack => {
                        const stateName =
                            stackState(
                                stack
                            );

                        return (
                            stateName ===
                                "held" ||
                            stateName ===
                                "issue"
                        );
                    }
                ).length
        };
    }

    function recordMatches(
        record,
        search
    ) {
        if (!search) {
            return true;
        }

        const haystack =
            upper(
                [
                    record.container,
                    record.priority,
                    record.yardLocation,
                    record.vessel,
                    record.voyage,
                    record.stops,
                    record.status
                ].join(
                    " "
                )
            );

        return haystack.includes(
            search
        );
    }

    function stackMatches(
        stack,
        search
    ) {
        if (!search) {
            return true;
        }

        if (
            upper(
                `${stack.block} ${stack.label}`
            ).includes(
                search
            )
        ) {
            return true;
        }

        return stack.records.some(
            record =>
                recordMatches(
                    record,
                    search
                )
        );
    }

    function isoShort(
        iso
    ) {
        const value =
            clean(
                iso
            );

        if (
            value.startsWith(
                "2"
            )
        ) {
            return "20'";
        }

        if (
            value.startsWith(
                "4"
            )
        ) {
            return "40'";
        }

        return value || "—";
    }

    function stateLabel(

        switch (

            case "clear-above":
                return "CLEAR ABOVE";

            case "top-of-sro":

            case "blocked":

            case "held":

    function containerLengthFeet(
        iso
    ) {
        const value =
            upper(
                iso
            );

        if (
            value.startsWith(
                "4"
            )
        ) {
            return 40;
        }

        if (
            value.startsWith(
                "2"
            )
        ) {
            return 20;
        }

        return 20;
    }

    function containerFootprintCells(
        iso
    ) {
        return containerLengthFeet(
            iso
        ) === 40
            ? 4
            : 2;
    }

    function stackFootprintCells(
        stack
    ) {
        if (
            !stack?.records?.length
        ) {
            return 2;
        }

        return Math.max(
            2,
            ...stack.records.map(
                record =>
                    containerFootprintCells(
                        record.iso
                    )
            )
        );
    }

    function dominantFootprintCells(
        yMap
    ) {
        const counts =
            new Map();

        for (
            const stack of
            yMap.values()
        ) {
            const cells =
                stackFootprintCells(
                    stack
                );

            counts.set(
                cells,
                (
                    counts.get(
                        cells
                    ) || 0
                ) + 1
            );
        }

        return Array.from(
            counts.entries()
        )
            .sort(
                (
                    a,
                    b
                ) =>
                    b[1] - a[1]
            )[0]?.[0] ||
            2;
    }

    function positionFootprints(
        yMap,
        positions
    ) {
        const fallback =
            dominantFootprintCells(
                yMap
            );

        return positions.map(
            position => {
                const stack =
                    yMap.get(
                        position
                    );

                return stack
                    ? stackFootprintCells(
                        stack
                    )
                    : fallback;
            }
        );
    }

    function footprintOffsetCells(
        footprints,
        index
    ) {
        return footprints
            .slice(
                0,
                Math.max(
                    0,
                    index
                )
            )
            .reduce(
                (
                    total,
                    cells
                ) =>
                    total + cells,
                0
            );
    }

    function snapGridPx(
        value
    ) {
        return Math.round(
            Number(value) /
            DPW_GRID_UNIT_PX
        ) *
        DPW_GRID_UNIT_PX;
    }

    function percentToGridPx(
        percent,
        totalPx
    ) {
        return snapGridPx(
            (
                Number(percent) /
                100
            ) *
            totalPx
        );
    }

    function blockLaneMetrics(
        stacks
    ) {
        if (
            !stacks?.size
        ) {
            return {
                totalCells:
                    12,
                widthPx:
                    12 *
                    DPW_GRID_UNIT_PX
            };
        }

        const groups =
            blockXGroups(
                stacks
            );

        let maxCells =
            12;

        for (
            const yMap of
            groups.values()
        ) {
            const positions =
                dpwHorizontalPositions(
                    yMap
                );

            const footprints =
                positionFootprints(
                    yMap,
                    positions
                );

            maxCells =
                Math.max(
                    maxCells,
                    footprints.reduce(
                        (
                            total,
                            cells
                        ) =>
                            total + cells,
                        0
                    )
                );
        }

        return {
            totalCells:
                maxCells,
            widthPx:
                maxCells *
                DPW_GRID_UNIT_PX
        };
    }

    function laneVisualHeightPx(
        stacks,
        fallbackHeightPx
    ) {
        /*
         * HYBRID COMPACT LOOK:
         * Keep the accurate 10ft grid scale, but make each FRT outline hug
         * the actual live DPW rows much more closely like the nicer compact
         * Transfocus-style versions.
         *
         * 1 visible container row = 1 grid-cell high.
         * No large empty vertical padding box around short lanes.
         */
        const groups =
            stacks?.size
                ? blockXGroups(
                    stacks
                )
                : new Map();

        const rowCount =
            Math.max(
                1,
                groups.size || 1
            );

        const compactHeightPx =
            rowCount * DPW_GRID_UNIT_PX;

        return Math.max(
            DPW_GRID_UNIT_PX,
            Math.min(
                fallbackHeightPx,
                compactHeightPx
            )
        );
    }

    function verticalLaneMetrics(
        stacks
    ) {
        const all =
            stacks?.size
                ? Array.from(
                    stacks.values()
                )
                : [];

        const positions =
            Array.from(
                new Set(
                    all
                        .map(
                            stack =>
                                Number(
                                    stack.y
                                )
                        )
                        .filter(
                            Number.isFinite
                        )
                )
            )
                .sort(
                    (a,b) =>
                        a - b
                );

        const maxFootprintCells =
            Math.max(
                2,
                ...all.map(
                    stack =>
                        stackFootprintCells(
                            stack
                        )
                )
            );

        return {
            positions,
            positionCount:
                Math.max(
                    1,
                    positions.length
                ),
            requiredHeightPx:
                Math.max(
                    DPW_GRID_UNIT_PX,
                    positions.length *
                    DPW_GRID_UNIT_PX
                ),
            requiredWidthPx:
                maxFootprintCells *
                DPW_GRID_UNIT_PX
        };
    }

    function physicalLayoutBox(
        layout,
        stacks
    ) {
        const leftPx =
            percentToGridPx(
                layout.left,
                DPW_STAGE_WIDTH_PX
            );

        let topPx =
            percentToGridPx(
                layout.top,
                DPW_STAGE_HEIGHT_PX
            );

        let widthPx =
            percentToGridPx(
                layout.width,
                DPW_STAGE_WIDTH_PX
            );

        let heightPx =
            percentToGridPx(
                layout.height,
                DPW_STAGE_HEIGHT_PX
            );

        if (
            layout.shape ===
                "lane"
        ) {
            const laneMetrics =
                blockLaneMetrics(
                    stacks
                );

            /*
             * Keep the real physical lane LENGTH from the DPW layout.
             * Only grow wider if the live container footprints ever exceed it.
             *
             * Visually though, match the nicer older version by making the
             * lane outlines hug the Transfocus-style tiles more closely,
             * instead of leaving a very tall empty box around them.
             */
            widthPx =
                Math.max(
                    widthPx,
                    laneMetrics.widthPx
                );

            const originalHeightPx =
                Math.max(
                    DPW_GRID_UNIT_PX,
                    heightPx
                );

            const compactHeightPx =
                laneVisualHeightPx(
                    stacks,
                    originalHeightPx
                );

            topPx +=
                Math.max(
                    0,
                    originalHeightPx - compactHeightPx
                );

            heightPx =
                compactHeightPx;
        }

        if (
            layout.shape ===
                "vertical-lane"
        ) {
            const metrics =
                verticalLaneMetrics(
                    stacks
                );

            heightPx =
                Math.max(
                    heightPx,
                    metrics.requiredHeightPx
                );

            widthPx =
                Math.max(
                    widthPx,
                    metrics.requiredWidthPx +
                    (2 * DPW_GRID_UNIT_PX)
                );
        }

        return {
            leftPx,
            topPx,
            widthPx:
                Math.max(
                    2 *
                    DPW_GRID_UNIT_PX,
                    widthPx
                ),
            heightPx:
                Math.max(
                    layout.shape === "lane"
                        ? DPW_GRID_UNIT_PX
                        : 2 * DPW_GRID_UNIT_PX,
                    heightPx
                )
        };
    }

    function laneStackPoint(
        stack,
        stacks
    ) {
        const groups =
            blockXGroups(
                stacks
            );

        const xValues =
            Array.from(
                groups.keys()
            )
                .sort(
                    (
                        a,
                        b
                    ) =>
                        a - b
                );

        const stackX =
            Number(
                stack.x
            );

        const yMap =
            groups.get(
                stackX
            ) ||
            new Map();

        const positions =
            dpwHorizontalPositions(
                yMap
            );

        const footprints =
            positionFootprints(
                yMap,
                positions
            );

        const position =
            Number(
                stack.y
            );

        const index =
            Math.max(
                0,
                positions.indexOf(
                    position
                )
            );

        const offsetCells =
            footprintOffsetCells(
                footprints,
                index
            );

        const footprintCells =
            footprints[
                index
            ] ||
            stackFootprintCells(
                stack
            );

        const rowIndex =
            Math.max(
                0,
                xValues.indexOf(
                    stackX
                )
            );

        const rowCount =
            Math.max(
                1,
                xValues.length
            );

        return {
            xPx:
                offsetCells *
                DPW_GRID_UNIT_PX,

            /*
             * Match Transfocus vertical cell behaviour:
             * row 0 sits on the BOTTOM grid row, additional physical rows
             * step upward exactly one 10ft grid cell at a time.
             * No more floating/centred containers inside the FRT outline.
             */
            bottomPx:
                rowIndex *
                DPW_GRID_UNIT_PX,

            widthPx:
                footprintCells *
                DPW_GRID_UNIT_PX,
            footprintCells,
            lane:
                true
        };
    }

    function verticalLaneStackPoint(
        stack,
        stacks
    ) {
        const metrics =
            verticalLaneMetrics(
                stacks
            );

        const y =
            Number(
                stack.y
            );

        const index =
            Math.max(
                0,
                metrics.positions.indexOf(
                    y
                )
            );

        const footprintCells =
            stackFootprintCells(
                stack
            );

        return {
            vertical:
                true,
            yPx:
                index *
                DPW_GRID_UNIT_PX,
            widthPx:
                footprintCells *
                DPW_GRID_UNIT_PX,
            footprintCells,
            lane:
                false
        };
    }

    function blockStackPoint(
        stack,
        stacks
    ) {
        const all =
            Array.from(
                stacks.values()
            );

        const xs =
            all
                .map(
                    item =>
                        Number(
                            item.x
                        )
                )
                .filter(
                    Number.isFinite
                );

        const ys =
            all
                .map(
                    item =>
                        Number(
                            item.y
                        )
                )
                .filter(
                    Number.isFinite
                );

        const minX =
            Math.min(
                ...xs
            );

        const maxX =
            Math.max(
                ...xs
            );

        const minY =
            Math.min(
                ...ys
            );

        const maxY =
            Math.max(
                ...ys
            );

        const x =
            Number(
                stack.x
            );

        const y =
            Number(
                stack.y
            );

        const nx =
            maxX === minX
                ? 0.10
                : (
                    0.08 +
                    (
                        (
                            x -
                            minX
                        ) /
                        (
                            maxX -
                            minX
                        )
                    ) *
                    0.68
                );

        const ny =
            maxY === minY
                ? 0.16
                : (
                    0.12 +
                    (
                        (
                            y -
                            minY
                        ) /
                        (
                            maxY -
                            minY
                        )
                    ) *
                    0.62
                );

        return {
            x:
                Math.max(
                    0.06,
                    Math.min(
                        0.76,
                        nx
                    )
                ),
            y:
                Math.max(
                    0.08,
                    Math.min(
                        0.72,
                        ny
                    )
                ),
            lane:
                false
        };
    }

    function stackPointForLayout(
        stack,
        stacks,
        layout
    ) {
        if (
            layout?.shape ===
                "lane" ||
            layout?.shape ===
                "small"
        ) {
            return laneStackPoint(
                stack,
                stacks
            );
        }

        if (
            layout?.shape ===
                "vertical-lane"
        ) {
            return verticalLaneStackPoint(
                stack,
                stacks
            );
        }

        return blockStackPoint(
            stack,
            stacks
        );
    }

    function renderRealBlock(
        block,
        stacks,
        search,
        ghost = false
    ) {
        const layout =
            REAL_DPW_BLOCK_LAYOUT[
                block
            ];

        if (!layout) {
            return "";
        }

        const hasData =
            Boolean(
                stacks &&
                stacks.size
            );

        let visibleStacks =
            [];

        if (hasData) {
            visibleStacks =
                Array.from(
                    stacks.values()
                )
                    .filter(
                        stack =>
                            !search ||
                            stackMatches(
                                stack,
                                search
                            )
                    );
        }

        if (
            search &&
            !upper(
                block
            ).includes(
                search
            ) &&
            !visibleStacks.length
        ) {
            return "";
        }

        const box =
            physicalLayoutBox(
                layout,
                stacks
            );

        const stackHtml =
            visibleStacks
                .map(
                    stack => {
                        const point =
                            stackPointForLayout(
                                stack,
                                stacks,
                                layout
                            );

                        const verticalLeftPx =
                            point.vertical
                                ? Math.max(
                                    0,
                                    Math.round(
                                        (
                                            (box.widthPx / DPW_GRID_UNIT_PX) -
                                            point.footprintCells
                                        ) / 2
                                    ) * DPW_GRID_UNIT_PX
                                )
                                : 0;

                        const stackStyle =
                            point.vertical
                                ? `left:${verticalLeftPx}px; top:${point.yPx}px; width:${point.widthPx}px;`
                                : point.lane
                                    ? `left:${point.xPx}px; bottom:${point.bottomPx}px; width:${point.widthPx}px;`
                                    : `left:${point.x * 100}%; top:${point.y * 100}%;`;

                        return `
                            <div
                                class="jamie-real-dpw-stack"
                                data-lane="${point.lane ? "true" : "false"}"
                                data-vertical="${point.vertical ? "true" : "false"}"
                                style="${stackStyle}"
                            >
                                ${renderTinyStackTile(
                                    stack,
                                    "is-mini"
                                )}
                            </div>
                        `;
                    }
                )
                .join("");

        return `
            <section
                class="jamie-real-dpw-block"
                data-has-sro="${hasData ? "true" : "false"}"
                data-shape="${escapeHtml(
                    layout.shape ||
                    "block"
                )}"
                style="
                    left:${box.leftPx}px;
                    top:${box.topPx}px;
                    width:${box.widthPx}px;
                    height:${box.heightPx}px;
                "
            >
                <button
                    type="button"
                    class="jamie-real-dpw-block-label"
                    ${hasData ? `data-open-block="${escapeHtml(block)}"` : ""}
                    ${hasData ? "" : "disabled"}
                >
                    ${escapeHtml(
                        layout.displayLabel ||
                        block.replace(
                            /^FRT-/,
                            ""
                        )
                    )}
                </button>

                ${stackHtml}
            </section>
        `;
    }

    function renderYardOverview(
        blocks,
        search
    ) {
        const allKnownBlocks =
            Array.from(
                new Set([
                    ...Object.keys(
                        REAL_DPW_BLOCK_LAYOUT
                    ),
                    ...Array.from(
                        blocks.keys()
                    )
                ])
            )
                .sort(
                    naturalCompare
                );

        const html =
            allKnownBlocks
                .map(
                    block =>
                        renderRealBlock(
                            block,
                            blocks.get(
                                block
                            ) ||
                            null,
                            search,
                            !blocks.has(
                                block
                            )
                        )
                )
                .filter(Boolean)
                .join("");

        if (!html) {
            return `
                <div class="jamie-yard-empty-message">
                    No DPW yard blocks match this search.
                </div>
            `;
        }

        return `
            <div class="jamie-real-dpw-yard">
                <div class="jamie-real-dpw-site-note">
                    DP WORLD ROUS HEAD — PHYSICAL YARD LAYOUT
                </div>

                ${html}
            </div>
        `;
    }

    function stackRecordsByTier(
        stack
    ) {
        return new Map(
            stack.records.map(
                record => [
                    record.location
                        .tier,
                    record
                ]
            )
        );
    }

    function highestKnownTier(
        stack
    ) {
        if (
            !stack ||
            !stack.records?.length
        ) {
            return 0;
        }

        return Math.max(
            ...stack.records.map(
                record =>
                    Number(
                        record.location
                            .tier
                    ) || 0
            )
        );
    }

    function stackHasGapBelowTop(
        stack
    ) {
        const highest =
            highestKnownTier(
                stack
            );

        if (highest <= 1) {
            return false;
        }

        const byTier =
            stackRecordsByTier(
                stack
            );

        for (
            let tier = 1;
            tier < highest;
            tier += 1
        ) {
            if (!byTier.has(tier)) {
                return true;
            }
        }

        return false;
    }

    function stackTopSroRecord(
        stack
    ) {
        return stack?.records
            ?.slice()
            .sort(
                (
                    a,
                    b
                ) =>
                    b.location.tier -
                    a.location.tier
            )[0] || null;
    }

    function blockXGroups(
        stacks
    ) {
        const groups =
            new Map();

        if (
            !stacks ||
            typeof stacks.values !==
                "function"
        ) {
            return groups;
        }

        for (
            const stack of
            stacks.values()
        ) {
            const x =
                Number(
                    stack.x
                );

            if (!groups.has(x)) {
                groups.set(
                    x,
                    new Map()
                );
            }

            groups
                .get(x)
                .set(
                    Number(
                        stack.y
                    ),
                    stack
                );
        }

        return groups;
    }

    function dpwHorizontalPositions(
        yMap
    ) {
        const observed =
            Array.from(
                yMap.keys()
            )
                .filter(
                    Number.isFinite
                );

        const maxObserved =
            observed.length
                ? Math.max(
                    ...observed
                )
                : 0;

        /*
         * The DPW stack viewer supplied by the user uses 01 -> 06.
         * Keep six positions as the normal view, but expand if a future
         * DPW location uses a position greater than 06.
         */
        const count =
            Math.max(
                6,
                maxObserved
            );

        return Array.from(
            {
                length:
                    count
            },
            (
                _,
                index
            ) =>
                index + 1
        );
    }

    function twoDigit(
        value
    ) {
        return String(
            value
        ).padStart(
            2,
            "0"
        );
    }

    function containerDisplayLines(
        container
    ) {
        const value =
            clean(
                container
            );

        if (value.length <= 6) {
            return {
                top:
                    value,
                bottom:
                    ""
            };
        }

        return {
            top:
                value.slice(
                    0,
                    4
                ),
            bottom:
                value.slice(4)
        };
    }

    function coordinateValues(
        stacks
    ) {
        const items =
            Array.from(
                stacks.values()
            );

        const xs =
            Array.from(
                new Set(
                    items.map(
                        stack =>
                            Number(
                                stack.x
                            )
                    )
                )
            )
                .filter(
                    Number.isFinite
                )
                .sort(
                    (
                        a,
                        b
                    ) =>
                        a - b
                );

        const ys =
            Array.from(
                new Set(
                    items.map(
                        stack =>
                            Number(
                                stack.y
                            )
                    )
                )
            )
                .filter(
                    Number.isFinite
                )
                .sort(
                    (
                        a,
                        b
                    ) =>
                        a - b
                );

        return {
            xs,
            ys
        };
    }

    function stackMapPosition(
        stack,
        coords
    ) {
        const x =
            Number(
                stack.x
            );

        const y =
            Number(
                stack.y
            );

        const xIndex =
            Math.max(
                0,
                coords.xs
                    .indexOf(
                        x
                    )
            );

        const yIndex =
            Math.max(
                0,
                coords.ys
                    .indexOf(
                        y
                    )
            );

        const xCount =
            Math.max(
                1,
                coords.xs.length
            );

        const yCount =
            Math.max(
                1,
                coords.ys.length
            );

        /*
         * Physical-looking placement based on the unique X/Y values present
         * in the current DPW block.
         */
        const left =
            8 +
            xIndex *
            (
                84 /
                xCount
            );

        const top =
            10 +
            yIndex *
            (
                76 /
                yCount
            );

        const width =
            Math.min(
                16,
                Math.max(
                    8,
                    68 /
                    xCount
                )
            );

        const height =
            Math.min(
                18,
                Math.max(
                    10,
                    58 /
                    yCount
                )
            );

        return {
            left,
            top,
            width,
            height
        };
    }

    function renderBlockMap(
        block,
        stacks,
        search
    ) {
        const filteredStacks =
            new Map(
                Array.from(
                    stacks.entries()
                ).filter(
                    (
                        [
                            ,
                            stack
                        ]
                    ) =>
                        !search ||
                        stackMatches(
                            stack,
                            search
                        )
                )
            );

        if (!filteredStacks.size) {
            return `
                <div class="jamie-yard-empty-message">
                    No stacks in ${escapeHtml(
                        block
                    )} match this search.
                </div>
            `;
        }

        const groups =
            blockXGroups(
                filteredStacks
            );

        const groupHtml =
            Array.from(
                groups.entries()
            )
                .sort(
                    (
                        a,
                        b
                    ) =>
                        a[0] - b[0]
                )
                .map(
                    (
                        [
                            x,
                            yMap
                        ]
                    ) => {
                        const positions =
                            dpwHorizontalPositions(
                                yMap
                            );

                        const footprints =
                            positionFootprints(
                                yMap,
                                positions
                            );

                        const physicalColumns =
                            footprints
                                .map(
                                    cells =>
                                        `${cells * DPW_GRID_UNIT_PX}px`
                                )
                                .join(
                                    " "
                                );

                        const physicalWidthPx =
                            footprints.reduce(
                                (
                                    total,
                                    cells
                                ) =>
                                    total + cells,
                                0
                            ) *
                            DPW_GRID_UNIT_PX;

                        const topCandidates =
                            positions
                                .map(
                                    position => {
                                        const stack =
                                            yMap.get(
                                                position
                                            );

                                        if (!stack) {
                                            return "";
                                        }

                                        const top =
                                            stackTopSroRecord(
                                                stack
                                            );

                                        if (!top) {
                                            return "";
                                        }

                                        const held =
                                            top.accessState ===
                                                "held" ||
                                            top.accessState ===
                                                "issue";

                                        return `
                                            <button
                                                type="button"
                                                class="jamie-dpw-ltr-chip"
                                                data-stack="${escapeHtml(
                                                    stack.key
                                                )}"
                                                data-held="${held ? "true" : "false"}"
                                            >
                                                <b>
                                                    ${escapeHtml(
                                                        twoDigit(
                                                            position
                                                        )
                                                    )}
                                                </b>

                                                <span>
                                                    ${escapeHtml(
                                                        top.container
                                                    )}
                                                </span>

                                                <small>
                                                    T${escapeHtml(
                                                        top.location.tier
                                                    )}
                                                    ${held ? " • HELD" : ""}
                                                </small>
                                            </button>
                                        `;
                                    }
                                )
                                .filter(Boolean)
                                .join("");

                        const headers =
                            positions.map(
                                position => `
                                    <div class="jamie-dpw-grid-position-head">
                                        ${escapeHtml(
                                            twoDigit(
                                                position
                                            )
                                        )}
                                    </div>
                                `
                            ).join("");

                        const rows =
                            [];

                        for (
                            let tier =
                                DPW_MAX_TIER;
                            tier >= 1;
                            tier -= 1
                        ) {
                            const cells =
                                positions.map(
                                    position => {
                                        const stack =
                                            yMap.get(
                                                position
                                            );

                                        if (!stack) {
                                            return `
                                                <div
                                                    class="jamie-dpw-grid-cell is-no-sro-column"
                                                    title="No container from this SRO is shown at position ${escapeHtml(twoDigit(position))}."
                                                ></div>
                                            `;
                                        }

                                        const byTier =
                                            stackRecordsByTier(
                                                stack
                                            );

                                        const record =
                                            byTier.get(
                                                tier
                                            );

                                        const highest =
                                            highestKnownTier(
                                                stack
                                            );

                                        if (!record) {
                                            const belowKnownTop =
                                                tier < highest;

                                            return `
                                                <button
                                                    type="button"
                                                    class="jamie-dpw-grid-cell is-missing"
                                                    data-stack="${escapeHtml(
                                                        stack.key
                                                    )}"
                                                    data-kind="${belowKnownTop ? "gap-below" : "above"}"
                                                    title="${escapeHtml(
                                                        belowKnownTop
                                                            ? "No SRO record at this tier even though a container is recorded above it. Something/unknown data must support the upper container."
                                                            : "No container from this SRO is shown at this tier. It may be empty or may belong to another movement/company."
                                                    )}"
                                                >
                                                    <span>
                                                        ${belowKnownTop ? "?" : ""}
                                                    </span>
                                                </button>
                                            `;
                                        }

                                        const display =
                                            containerDisplayLines(
                                                record.container
                                            );

                                        const held =
                                            record.accessState ===
                                                "held" ||
                                            record.accessState ===
                                                "issue";

                                        const isTop =
                                            tier === highest;

                                        const clearAbove =
                                            record.accessState ===
                                                "clear-above";

                                        return `
                                            <button
                                                type="button"
                                                class="jamie-dpw-grid-cell is-sro-container"
                                                data-stack="${escapeHtml(
                                                    stack.key
                                                )}"
                                                data-held="${held ? "true" : "false"}"
                                                data-top-sro="${isTop ? "true" : "false"}"
                                                data-clear-above="${clearAbove ? "true" : "false"}"
                                                title="${escapeHtml(
                                                    `${record.container} • ${record.yardLocation} • Priority ${record.priority || "—"}${held ? " • HELD" : ""}`
                                                )}"
                                            >
                                                <span class="jamie-dpw-grid-container-prefix">
                                                    ${escapeHtml(
                                                        display.top
                                                    )}
                                                </span>

                                                <span class="jamie-dpw-grid-container-number">
                                                    ${escapeHtml(
                                                        display.bottom
                                                    )}
                                                </span>

                                                <small>
                                                    P${escapeHtml(
                                                        record.priority ||
                                                        "—"
                                                    )}
                                                    ${held ? " • HELD" : ""}
                                                </small>
                                            </button>
                                        `;
                                    }
                                ).join("");

                            rows.push(`
                                <div class="jamie-dpw-grid-tier-label">
                                    ${tier}
                                </div>
                                ${cells}
                                <div class="jamie-dpw-grid-tier-label is-right">
                                    ${tier}
                                </div>
                            `);
                        }

                        return `
                            <section class="jamie-dpw-bay-panel">
                                <div class="jamie-dpw-bay-head">
                                    <div>
                                        <span>
                                            ${escapeHtml(
                                                block
                                            )}
                                        </span>

                                        <strong>
                                            STACK ROW ${escapeHtml(
                                                twoDigit(
                                                    x
                                                )
                                            )}
                                        </strong>
                                    </div>

                                    <div class="jamie-dpw-direction">
                                        WORK LEFT
                                        <b>→</b>
                                        RIGHT
                                    </div>
                                </div>

                                <div class="jamie-dpw-ltr-strip">
                                    <span class="jamie-dpw-ltr-title">
                                        TOP SRO CONTAINERS — LEFT → RIGHT
                                    </span>

                                    <div class="jamie-dpw-ltr-list">
                                        ${topCandidates || `
                                            <span class="jamie-yard-access-none">
                                                No SRO containers in this row.
                                            </span>
                                        `}
                                    </div>
                                </div>

                                <div
                                    class="jamie-dpw-stack-grid"
                                    style="
                                        --dpw-physical-columns:${physicalColumns};
                                        --dpw-physical-width:${physicalWidthPx}px;
                                        grid-template-columns: 28px ${physicalColumns} 28px;
                                    "
                                >
                                    <div class="jamie-dpw-grid-corner"></div>
                                    ${headers}
                                    <div class="jamie-dpw-grid-corner"></div>

                                    ${rows.join("")}
                                </div>

                                <div class="jamie-dpw-grid-ground">

    function renderStackPopup(
        stack
    ) {
        if (!stack) {
            return "";
        }

        const sorted =
            stack.records
                .slice()
                .sort(
                    (
                        a,
                        b
                    ) =>
                        b.location.tier -
                        a.location.tier
                );

        const byTier =
            stackRecordsByTier(
                stack
            );

        const highest =
            highestKnownTier(
                stack
            );

        const hasGapBelow =
            stackHasGapBelowTop(
                stack
            );

        const tiers =
            [];

        for (
            let tier =
                DPW_MAX_TIER;
            tier >= 1;
            tier -= 1
        ) {
            const record =
                byTier.get(
                    tier
                );

            if (!record) {
                const belowKnownTop =
                    tier < highest;

                tiers.push(`
                    <div
                        class="jamie-yard-popup-tier is-unknown-other"
                        data-kind="${belowKnownTop ? "gap-below" : "above"}"
                    >
                        <span class="jamie-yard-popup-tier-number">
                            ${tier}
                        </span>

                        <div class="jamie-yard-popup-unknown">
                            <strong>
                                ${belowKnownTop ? "UNKNOWN BELOW" : "NO SRO DATA ABOVE"}
                            </strong>

                            <span>
                                ${belowKnownTop
                                    ? "A container is recorded above this tier, so this cannot be assumed empty."
                                    : "May be physically empty, or may contain a container outside this SRO."}
                            </span>
                        </div>
                    </div>
                `);

                continue;
            }

            const held =
                record.accessState ===
                    "held";

            const issue =
                record.accessState ===
                    "issue";

            const popupState =
                held
                    ? "held"
                    : (
                        issue
                            ? "issue"
                            : "normal"
                    );

            const statusText =
                held
                    ? "HELD"
                    : (
                        issue
                            ? (
                                clean(
                                    record.status
                                ) ||
                                "ISSUE"
                            )
                            : ""
                    );

            tiers.push(`
                <div
                    class="jamie-yard-popup-tier"
                    data-popup-state="${escapeHtml(
                        popupState
                    )}"
                >
                    <span class="jamie-yard-popup-tier-number">
                        ${tier}
                    </span>

                    <div class="jamie-yard-popup-container-main">
                        <strong
                            title="Double-click or drag to copy container number"
                        >
                            ${escapeHtml(
                                record.container
                            )}
                        </strong>

                        <span>
                            Priority ${escapeHtml(
                                record.priority ||
                                "—"
                            )}
                            •
                            ${escapeHtml(
                                isoShort(
                                    record.iso
                                )
                            )}
                        </span>
                    </div>

                    <div class="jamie-yard-popup-tier-actions">
                        ${statusText
                            ? `
                                <div class="jamie-yard-popup-held-label">
                                    ${escapeHtml(
                                        statusText
                                    )}
                                </div>
                            `
                            : ""
                        }

                        <button
                            type="button"
                            class="jamie-yard-popup-row-button"
                            data-jump-container="${escapeHtml(
                                record.container
                            )}"
                            title="Go to this container in the DPW SRO list"
                        >
                            GO TO ROW
                        </button>
                    </div>
                </div>
            `);
        }

        const top =
            sorted[0] ||
            null;

        let confidenceText =
            "NO SRO CONTAINERS";

        if (top) {
            if (
                highest ===
                    DPW_MAX_TIER
            ) {
                confidenceText =
                    "TIER 4 — NOTHING CAN BE ABOVE";
            } else if (hasGapBelow) {
                confidenceText =
                    "GAP BELOW — SUPPORT IS NOT SHOWN IN THIS SRO";
            } else {
                confidenceText =
                    `SRO TIERS 1–${highest} CONTIGUOUS — ABOVE IS NOT SHOWN`;
            }
        }

        return `
            <div class="jamie-yard-stack-popup-backdrop">
                <div class="jamie-yard-stack-popup">
                    <div class="jamie-yard-stack-popup-head">
                        <div>
                            <span>
                                ${escapeHtml(
                                    stack.block
                                )}
                            </span>

                            <strong>
                                STACK ${escapeHtml(
                                    stack.label
                                )}
                            </strong>
                        </div>

                        <button
                            type="button"
                            class="jamie-yard-popup-close"
                            data-action="close-stack"
                            title="Close stack"
                            aria-label="Close stack"
                        >
                            ×
                        </button>
                    </div>

                    <div class="jamie-yard-stack-popup-summary">
                        <span>
                            ${stack.records.length}
                            SRO container${stack.records.length === 1 ? "" : "s"}
                        </span>

                        <span>
                            Highest SRO:
                            <b>
                                ${escapeHtml(
                                    top?.container ||
                                    "—"
                                )}
                            </b>
                        </span>
                    </div>

                    <div class="jamie-yard-stack-popup-tiers">
                        <div class="jamie-yard-popup-top-label">
                            TOP OF STACK — MAX TIER 4
                        </div>

                        ${tiers.join("")}

                        <div class="jamie-yard-popup-ground">
                            GROUND
                        </div>
                    </div>

                    <div class="jamie-yard-stack-popup-foot">
                        Missing ABOVE the highest SRO tier is not automatically another company. Missing BELOW a known upper container cannot be assumed empty.
                    </div>
                </div>
            </div>
        `;
    }

    function findStack(
        blocks,
        key
    ) {
        if (!key) {
            return null;
        }

        for (
            const stacks of
            blocks.values()
        ) {
            if (
                stacks.has(
                    key
                )
            ) {
                return stacks.get(
                    key
                );
            }
        }

        return null;
    }

    function renderAccessibleStrip(
        records
    ) {
        const search =
            upper(
                state.search
            );

        const clearAbove =
            records
                .filter(
                    record =>
                        record.accessState ===
                            "clear-above" &&
                        (
                            state.view !==
                                "BLOCK" ||
                            record.location
                                ?.block ===
                                state.selectedBlock
                        ) &&
                        recordMatches(
                            record,
                            search
                        )
                )
                .sort(
                    (
                        a,
                        b
                    ) =>
                        priorityNumber(
                            a.priority
                        ) -
                        priorityNumber(
                            b.priority
                        )
                );

        if (!clearAbove.length) {
            return `
                <span class="jamie-yard-access-none">
                    No Tier 4 container in this view.
                </span>
            `;
        }

        return clearAbove.map(
            record => `
                <button
                    type="button"
                    class="jamie-yard-access-chip"
                    data-container="${escapeHtml(
                        record.container
                    )}"
                >
                    <b>
                        P${escapeHtml(
                            record.priority ||
                            "—"
                        )}
                    </b>

                    <span>
                        ${escapeHtml(
                            record.container
                        )}
                    </span>

                    <small>
                        T4 • ${escapeHtml(
                            record.yardLocation
                        )}
                    </small>
                </button>
            `
        ).join(
            ""
        );
    }

    function ensurePanel() {

        const holder =
            getHolder();

        if (!holder) {
            return null;
        }

        let panel =
            document.getElementById(
                PANEL_ID

        if (!panel) {
            panel =
                document.createElement(
                    "section"
                );

            panel.id =
                PANEL_ID;

            panel.innerHTML = `
                <div class="yard-map-workspace">
                    <div class="yard-map-toolbar">
                        <div class="yard-map-toolbar__content">

                                <input

                    panel,

                panel
            );

                panel.querySelector(

                );

            search?.addEventListener(
                "input",
                () => {
                    state.search =
                        upper(
                            search.value
                        );

                    render(
                        true
                    );
                }
            );

            panel.addEventListener(
                "click",
                event => {
                    const target =
                        event.target;

                    if (
                        !(
                            target instanceof
                            Element
                        )
                    ) {
                        return;
                    }

                    const action =
                        target.closest(
                            "[data-action]"
                        )
                            ?.getAttribute(
                                "data-action"
                            );

                    if (
                        action ===
                            "refresh"
                    ) {
                        state.lastSignature =
                            "";

                        render(
                            true
                        );

                        return;
                    }

                    if (
                        action ===
                            "close-map"
                    ) {
                        closePanel();

                        return;
                    }

                    if (
                        action ===
                            "close-stack"
                    ) {
                        state.selectedStackKey =
                            "";

                        render(
                            true
                        );

                        return;
                    }

                    if (
                        action ===
                            "back-yard"
                    ) {
                        state.view =
                            "YARD";

                        state.selectedBlock =
                            "";

                        state.selectedStackKey =
                            "";

                        render(
                            true
                        );

                        return;
                    }

                    const viewButton =
                        target.closest(
                            "[data-view]"
                        );

                    if (viewButton) {
                        const view =
                            clean(
                                viewButton.getAttribute(
                                    "data-view"
                                )
                            );

                        if (
                            view ===
                                "YARD"
                        ) {
                            state.view =
                                "YARD";

                            state.selectedBlock =
                                "";

                            state.selectedStackKey =
                                "";
                        } else {
                            state.view =
                                "BLOCK";

                            state.selectedBlock =
                                clean(
                                    viewButton.getAttribute(
                                        "data-block"
                                    )
                                );

                            state.selectedStackKey =
                                "";
                        }

                        render(
                            true
                        );

                        return;
                    }

                    const blockShape =
                        target.closest(
                            "[data-open-block]"
                        );

                    if (blockShape) {
                        state.view =
                            "BLOCK";

                        state.selectedBlock =
                            clean(
                                blockShape.getAttribute(
                                    "data-open-block"
                                )
                            );

                        state.selectedStackKey =
                            "";

                        render(
                            true
                        );

                        return;
                    }

                    const stackShape =
                        target.closest(
                            "[data-stack]"
                        );

                    if (stackShape) {
                        state.selectedStackKey =
                            clean(
                                stackShape.getAttribute(
                                    "data-stack"
                                )
                            );

                        render(
                            true
                        );

                        return;
                    }

                    const rowButton =
                        target.closest(
                            "[data-jump-container]"
                        );

                    if (rowButton) {
                        focusRow(
                            clean(
                                rowButton.getAttribute(
                                    "data-jump-container"
                                )
                            )
                        );

                        return;
                    }

                    /*
                     * Keep the green Tier-4 strip behaviour, but DO NOT make
                     * container text inside the opened stack popup navigate.
                     * That text stays selectable/copyable.
                     */
                    const containerButton =
                        target.closest(
                            "[data-container]"
                        );

                    if (
                        containerButton &&
                        !containerButton.closest(
                            ".jamie-yard-stack-popup"
                        )
                    ) {
                        focusRow(
                            clean(
                                containerButton.getAttribute(
                                    "data-container"
                                )
                            )
                        );
                    }
                }
            );
        }

        if (
            panel.nextElementSibling !==
            holder
        ) {
            holder.parentElement
                ?.insertBefore(
                    panel,
                    holder
                );
        }

        panel.style.display =
            state.closed
                ? "none"
                : "";

        syncPanelViewportWidth(
            panel
        );

        return panel;
    }

    function focusRow(
        container
    ) {
        if (!container) {
            return false;
        }

        const record =
            readRecords()
                .find(
                    item =>
                        upper(
                            item.container
                        ) ===
                        upper(
                            container
                        )
                );

        if (
            !record?.rowElement
        ) {
            return false;
        }

        getHolder()
            ?.querySelectorAll(
                `tr[${ROW_FOCUS_ATTR}="true"]`
            )
            .forEach(
                row =>
                    row.removeAttribute(
                        ROW_FOCUS_ATTR
                    )
            );

        record.rowElement
            .setAttribute(
                ROW_FOCUS_ATTR,
                "true"
            );

        clearTimeout(
            state.focusTimer
        );

        state.focusTimer =
            setTimeout(
                () => {
                    try {
                        record.rowElement
                            ?.removeAttribute(
                                ROW_FOCUS_ATTR
                            );
                    } catch (_) {}
                },
                4500
            );

        try {
            record.rowElement
                .scrollIntoView({
                    behavior:
                        "smooth",
                    block:
                        "center",
                    inline:
                        "nearest"
                });
        } catch (_) {}

        return true;
    }

    function closePanel() {
        state.closed =
            true;

        const panel =
            document.getElementById(
                PANEL_ID
            );

        if (panel) {
            panel.style.display =
                "none";
        }
    }

    function openPanel() {
        state.closed =
            false;

        const panel =
            ensurePanel();

        if (panel) {
            panel.style.display =
                "";
        }
    }

    function syncPanelViewportWidth(
        panel = null
    ) {
        const root =
            panel ||
            document.getElementById(
                PANEL_ID
            );

        if (!root) {
            return false;
        }

        const rect =
            root.getBoundingClientRect();

        const leftPx =
            Math.max(
                0,
                rect.left || 0
            );

        const viewportWidth =
            Math.max(
                800,
                window.innerWidth ||
                document.documentElement.clientWidth ||
                0
            );

        /*
         * The native SRO table only occupies the left side of the page.
         * The Yard Map should NOT inherit that narrow table width.
         * Expand from its real left edge to the viewport's right margin.
         */
        const availableWidth =
            Math.max(
                1100,
                viewportWidth -
                leftPx -
                12
            );

        root.style.width =
            `${Math.floor(availableWidth)}px`;

        root.style.maxWidth =
            "none";

        const workspace =
            root.querySelector(
                ".yard-map-workspace"
            );

        if (workspace) {
            workspace.style.width =
                "100%";

            workspace.style.maxWidth =
                "none";
        }

        return true;
    }

    function applyStageZoom(
        panel = null
    ) {
        const root =
            panel ||
            document.getElementById(
                PANEL_ID
            );

        if (!root) {
            return false;
        }

        const host =
            root.querySelector(
                ".yard-map__stage-host"
            );

        const stage =
            root.querySelector(
                ".jamie-yard-stage"
            );

        if (
            !host ||
            !stage
        ) {
            return false;
        }

        state.zoom =
            clamp(
                state.zoom,
                state.minZoom,
                state.maxZoom
            );

        stage.style.zoom =
            String(
                state.zoom
            );

        stage.style.transform =
            "none";

        stage.style.transformOrigin =
            "top left";

        host.setAttribute(
            "title",
            `Left click drag = move around | Mouse wheel = zoom (${Math.round(state.zoom * 100)}%)`
        );

        return true;
    }

    function computeRenderedOverviewBounds(
        panel = null
    ) {
        const root =
            panel ||
            document.getElementById(
                PANEL_ID
            );

        if (!root) {
            return null;
        }

        const stage =
            root.querySelector(
                ".jamie-yard-stage"
            );

        if (!stage) {
            return null;
        }

        const blocks =
            Array.from(
                stage.querySelectorAll(
                    ".jamie-real-dpw-block"
                )
            );

        if (!blocks.length) {
            return null;
        }

        let minLeft = Infinity;
        let minTop = Infinity;
        let maxRight = 0;
        let maxBottom = 0;

        blocks.forEach(block => {
            const left =
                block.offsetLeft || 0;
            const top =
                block.offsetTop || 0;
            const width =
                block.offsetWidth || 0;
            const height =
                block.offsetHeight || 0;

            minLeft = Math.min(
                minLeft,
                left
            );

            minTop = Math.min(
                minTop,
                top
            );

            maxRight = Math.max(
                maxRight,
                left + width
            );

            maxBottom = Math.max(
                maxBottom,
                top + height
            );
        });

        if (!Number.isFinite(minLeft) || !Number.isFinite(minTop)) {
            return null;
        }

        const margin = 22;

        return {
            left: Math.max(0, minLeft - margin),
            top: Math.max(0, minTop - margin),
            right: Math.min(DPW_STAGE_WIDTH_PX, maxRight + margin),
            bottom: Math.min(DPW_STAGE_HEIGHT_PX, maxBottom + margin),
            width: Math.min(DPW_STAGE_WIDTH_PX, Math.max(200, (maxRight - minLeft) + margin * 2)),
            height: Math.min(DPW_STAGE_HEIGHT_PX, Math.max(180, (maxBottom - minTop) + margin * 2))
        };
    }

    function computeAutoFitZoom(
        panel = null
    ) {
        const root =
            panel ||
            document.getElementById(
                PANEL_ID
            );

        if (!root) {
            return null;
        }

        const host =
            root.querySelector(
                ".yard-map__stage-host"
            );

        if (!host) {
            return null;
        }

        const bounds =
            computeRenderedOverviewBounds(
                root
            );

        const fitWidthPx =
            Math.max(
                640,
                (host.clientWidth || 0) - 22
            );

        const fitHeightPx =
            Math.max(
                420,
                (host.clientHeight || 0) - 18
            );

        const fitTargetWidth =
            bounds?.width ||
            DPW_STAGE_WIDTH_PX;

        const fitTargetHeight =
            bounds?.height ||
            DPW_STAGE_HEIGHT_PX;

        const rawFitZoom =
            Math.min(
                fitWidthPx / fitTargetWidth,
                fitHeightPx / fitTargetHeight
            );

        const readableFitZoom =
            Math.max(
                MIN_READABLE_OVERVIEW_ZOOM,
                (rawFitZoom || DEFAULT_OVERVIEW_ZOOM) *
                AUTO_FIT_READABILITY_BIAS
            );

        return clamp(
            Math.min(
                MAX_AUTO_FIT_ZOOM,
                readableFitZoom
            ),
            state.minZoom,
            state.maxZoom
        );
    }

    function autoFitOverview(

        panel = null,
        force = false
    ) {
        const root =
            panel ||
            document.getElementById(
                PANEL_ID
            );

        if (!root) {
            return false;
        }

        if (state.didAutoFit && !force) {
            return false;
        }

        const host =
            root.querySelector(
                ".yard-map__stage-host"
            );

        const zoom =
            computeAutoFitZoom(
                root
            );

        if (!host || !zoom) {
            return false;
        }

        state.zoom =
            zoom;

        applyStageZoom(
            root
        );

        host.scrollLeft =
            Math.round(
                18 * state.zoom
            );

        host.scrollTop =
            Math.round(
                8 * state.zoom
            );

        state.didAutoFit = true;

        return true;
    }

    function installStageInteractions(
        panel = null
    ) {
        const root =
            panel ||
            document.getElementById(
                PANEL_ID
            );

        if (!root) {
            return false;
        }

        const host =
            root.querySelector(
                ".yard-map__stage-host"
            );

        if (!host) {
            return false;
        }

        if (
            root.getAttribute(
                "data-pan-zoom-bound"
            ) ===
                "true"
        ) {
            return applyStageZoom(
                root
            );
        }

        root.setAttribute(
            "data-pan-zoom-bound",
            "true"
        );

        host.style.cursor =
            "grab";

        addManagedEvent(
            host,
            "wheel",
            event => {
                if (
                    !event ||
                    !(event.target instanceof Element)
                ) {
                    return;
                }

                if (
                    event.target.closest(
                        ".jamie-yard-stack-popup"
                    )
                ) {
                    return;
                }

                event.preventDefault();

                const delta =
                    event.deltaY < 0
                        ? 0.10
                        : -0.10;

                state.zoom =
                    clamp(
                        state.zoom + delta,
                        state.minZoom,
                        state.maxZoom
                    );

                applyStageZoom(
                    root
                );
            },
            {
                passive:
                    false
            }
        );

        addManagedEvent(
            host,
            "mousedown",
            event => {
                if (
                    event.button !== 0 ||
                    !(event.target instanceof Element)
                ) {
                    return;
                }

                if (
                    event.target.closest(
                        ".jamie-yard-stack-popup, button, input, textarea, select, a"
                    )
                ) {
                    return;
                }

                state.isDragging =
                    true;

                state.dragStartX =
                    event.clientX;

                state.dragStartY =
                    event.clientY;

                state.dragScrollLeft =
                    host.scrollLeft;

                state.dragScrollTop =
                    host.scrollTop;

                host.style.cursor =
                    "grabbing";

                event.preventDefault();
            }
        );

        addManagedEvent(
            window,
            "mousemove",
            event => {
                if (!state.isDragging) {
                    return;
                }

                host.scrollLeft =
                    state.dragScrollLeft -
                    (
                        event.clientX -
                        state.dragStartX
                    );

                host.scrollTop =
                    state.dragScrollTop -
                    (
                        event.clientY -
                        state.dragStartY
                    );
            }
        );

        const stopDragging =
            () => {
                if (!state.isDragging) {
                    return;
                }

                state.isDragging =
                    false;

                host.style.cursor =
                    "grab";
            };

        addManagedEvent(
            window,
            "mouseup",
            stopDragging
        );

        addManagedEvent(
            host,
            "mouseleave",
            () => {
                if (!state.isDragging) {
                    host.style.cursor =
                        "grab";
                }
            }
        );

        addManagedEvent(
            window,
            "blur",
            stopDragging
        );

        return applyStageZoom(
            root
        );
    }

    function render(
        force = false
    ) {
        const panel =
            ensurePanel();

        if (!panel) {
            return false;
        }

        const analysis =
            analyse(
                readRecords()
            );

        const records =
            analysis.records;

        const blocks =
            groupedData(
                records
            );

        const search =
            upper(
                state.search
            );

        const signature =
            JSON.stringify({
                view:
                    state.view,
                selectedBlock:
                    state.selectedBlock,
                selectedStackKey:
                    state.selectedStackKey,
                search:
                    state.search,
                rows:
                    records.map(
                        record => [
                            record.container,
                            record.priority,
                            record.yardLocation,
                            record.stops,
                            record.status,
                            record.selectable
                        ]
                    )
            });

        if (
            !force &&
            signature ===
                state.lastSignature
        ) {
            return true;
        }

        state.lastSignature =
            signature;

        const tabs =
            panel.querySelector(
                ".yard-map-tabs__list"
            );

        const stage =
            panel.querySelector(
                ".jamie-yard-stage"
            );


        const locatedCount =
            panel.querySelector(
                '[data-count="located"]'
            );


        const title =
            panel.querySelector(
                ".yard-map-toolbar__title"
            );

        if (
            !tabs ||
            !stage ||
            !locatedCount ||
            !title
        ) {
            return false;
        }

        const vessel =
            majorityVessel(
                records
            );

        title.textContent =
            vessel
                ? `DPW SRO Yard — ${vessel.vessel}${vessel.voyage ? ` ${vessel.voyage}` : ""}`
                : "DPW SRO Yard";

        locatedCount.textContent =
            String(
                records.filter(
                    record =>
                        record.location
                ).length
            );


        tabs.innerHTML =
            renderTabs(
                blocks
            );


        let mapHtml =
            "";

        if (
            state.view ===
                "BLOCK" &&
            blocks.has(
                state.selectedBlock
            )
        ) {
            mapHtml =
                renderBlockMap(
                    state.selectedBlock,
                    blocks.get(
                        state.selectedBlock
                    ),
                    search
                );
        } else {
            mapHtml =
                renderYardOverview(
                    blocks,
                    search
                );
        }

        const selectedStack =
            findStack(
                blocks,
                state.selectedStackKey
            );

        stage.innerHTML =
            mapHtml +
            renderStackPopup(
                selectedStack
            );

        syncPanelViewportWidth(
            panel
        );

        installStageInteractions(
            panel
        );

        if (
            !state.didAutoFit &&
            state.view === "YARD"
        ) {
            autoFitOverview(
                panel
            );
        } else {
            applyStageZoom(
                panel
            );
        }

        return true;
    }

    function scheduleRender(
        delay = 100
    ) {
        clearTimeout(
            state.renderTimer
        );

        state.renderTimer =
            setTimeout(
                () =>
                    render(
                        false
                    ),
                Math.max(
                    0,
                    Number(
                        delay
                    ) || 0
                )
            );
    }

    function attachObserver() {
        const holder =
            getHolder();

        if (!holder) {
            return false;
        }

        if (
            state.observedHolder ===
                holder &&
            state.observer
        ) {
            return true;
        }

        try {
            state.observer
                ?.disconnect();
        } catch (_) {}

        state.observedHolder =
            holder;

        state.observer =
            new MutationObserver(
                () =>
                    scheduleRender(
                        90
                    )
            );

        state.observer.observe(
            holder,
            {
                childList:
                    true,
                subtree:
                    true,
                characterData:
                    true
            }
        );

        return true;
    }

    function finiteRecovery() {
        [
            0,
            120,
            350,
            800
        ].forEach(
            delay =>
                setTimeout(
                    () => {
                        attachObserver();

                        if (
                            !state.closed
                        ) {
                            render(
                                false
                            );
                        }
                    },
                    delay
                )
        );
    }

    function installRecoveryEvents() {
        addManagedEvent(
            document,
            "click",
            event => {
                const target =
                    event.target;

                if (
                    !(
                        target instanceof
                        Element
                    )
                ) {
                    return;
                }

                const clickable =
                    target.closest(
                        "a, button, span, input"
                    );

                if (!clickable) {
                    return;
                }

                const href =
                    clean(
                        clickable.getAttribute(
                            "href"
                        )
                    );

                const id =
                    clean(
                        clickable.id
                    );

                const label =
                    upper(
                        clickable.textContent ||
                        clickable.getAttribute(
                            "value"
                        ) ||
                        clickable.getAttribute(
                            "title"
                        )
                    );

                if (
                    href ===
                        "#StackRunTabs3" ||
                    id ===
                        "refreshContainer" ||
                    label ===
                        "SRO" ||
                    label.includes(
                        "RE-VALIDATE CONTAINERS"
                    )
                ) {
                    setTimeout(
                        finiteRecovery,
                        50
                    );
                }
            },
            true
        );

        let resizeTimer =
            null;

        addManagedEvent(
            window,
            "resize",
            () => {
                clearTimeout(
                    resizeTimer
                );

                resizeTimer =
                    setTimeout(
                        () => {
                            if (
                                !state.closed
                            ) {
                                state.lastSignature =
                                    "";

                                syncPanelViewportWidth();

                                render(
                                    true
                                );
                            }
                        },
                        160
                    );
            },
            {
                passive:
                    true
            }
        );

        registerUnbinder(
            () =>
                clearTimeout(
                    resizeTimer
                )
        );

        const jq =
            window.jQuery ||
            window.$;

        if (
            jq &&
            typeof jq ===
                "function"
        ) {
            try {
                jq(
                    document
                ).on(
                    "ajaxComplete.jamieDpwSroYardMapV15",
                    () =>
                        setTimeout(
                            finiteRecovery,
                            100
                        )
                );

                registerUnbinder(
                    () => {
                        try {
                            jq(
                                document
                            ).off(
                                "ajaxComplete.jamieDpwSroYardMapV15"
                            );
                        } catch (_) {}
                    }
                );
            } catch (_) {}
        }
    }

    function cleanup() {
        clearTimeout(
            state.renderTimer
        );

        clearTimeout(
            state.focusTimer
        );

        try {
            state.observer
                ?.disconnect();
        } catch (_) {}

        state.observer =
            null;

        state.observedHolder =
            null;

        state.isDragging =
            false;

        try {
            state.unbinders
                .splice(0)
                .forEach(
                    fn => {
                        try {
                            fn();
                        } catch (_) {}
                    }
                );
        } catch (_) {}

        document
            .getElementById(
                PANEL_ID
            )
            ?.remove();

        document
            .getElementById(
                STYLE_ID
            )
            ?.remove();

        document
            .getElementById(
                "jamie-dpw-sro-yard-map-v4-transfocus-overrides"
            )
            ?.remove();

        delete window[
            GUARD
        ];
    }

    function start() {
        installStyle();
        installV3TransfocusLook();
        installRecoveryEvents();

        openPanel();
        finiteRecovery();

        [
            1200,
            2600
        ].forEach(
            delay =>
                setTimeout(
                    finiteRecovery,
                    delay
                )
        );

        console.info(
            `[${NAME}] active.`
        );
    }

    if (
        document.readyState ===
            "loading"
    ) {
        document.addEventListener(
            "DOMContentLoaded",
            start,
            {
                once:
                    true
            }
        );
    } else {
        start();
    }
})();
